1//! Der leniente Leser für Claude Codes Transkript — die reiche Zweitquelle.
2//!
3//! Der Hook liefert Zeitpunkt, Reihenfolge und Tool-Calls, aber nicht den
4//! Antworttext des Modells, nicht die Token-Zähler und nicht die Modell-ID. Die
5//! stehen nur im Transkript, einer JSONL-Datei (eine Zeile ein Ereignis). Dieses
6//! Modul liest genau diese drei Dinge heraus — mehr nicht.
7//!
8//! # Warum leniant und nicht streng
9//!
10//! Das Transkript gehört jemand anderem, und sein Format ändert sich zwischen
11//! Claude-Code-Versionen. Ein strenger Parser wäre eine Wette darauf, dass sich
12//! nichts ändert — eine Wette, die man verliert. Deshalb: Jede Zeile, die sich
13//! nicht deuten lässt, wird **übersprungen**, nie zum Fehler. Was wir verstehen,
14//! nehmen wir; der Rest ist Sache des Hooks, der ohnehin die verlässlichere
15//! Quelle für Struktur ist.
16//!
17//! # Was hier bewusst *nicht* passiert
18//!
19//! Kein Abgleich von Transkript-Einträgen mit Journal-Events über UUIDs. Der
20//! Adapter (siehe [`crate::adapter`]) ordnet die Assistant-Texte den
21//! Assistant-Zügen der Reihe nach zu — best effort. Ein UUID-genauer Abgleich
22//! wäre genauer, aber auch zerbrechlicher; er kann später additiv dazukommen,
23//! ohne dass sich der Vertrag dieses Moduls ändert.
24

28/// Was sich aus einem Transkript herausziehen lässt.
29#[derive(Debug, Clone, Default, PartialEq, Eq)]
30pub struct Transcript {
31 /// Das zuletzt gesehene Modell. `None`, wenn keine Assistant-Nachricht eine
32 /// Modell-ID trug.
33 pub model: Option<Model>,
34
35 /// Die aufsummierten Token-Zähler über alle Assistant-Nachrichten.
36 ///
37 /// `input_tokens` ist die Summe der Pro-Anfrage-Eingaben; weil jede Anfrage
38 /// den Kontext erneut sendet, ist das eine Obergrenze, kein Zähler
39 /// verschiedener Token. `output_tokens` summiert das tatsächlich Erzeugte.
40 pub usage: Usage,
41
42 /// Die zuletzt gesehene Agent-Version (Claude Codes `version`-Feld).
43 pub agent_version: Option<String>,
44
45 /// Die Assistant-Texte in Reihenfolge — je Assistant-Nachricht ein Eintrag,
46 /// leere übersprungen.
47 pub assistant_texts: Vec<String>,
48}

50/// Liest ein Transkript aus rohen JSONL-Bytes. Schlägt nie fehl: Was sich nicht
51/// deuten lässt, wird übersprungen.
52pub fn parse(bytes: &[u8]) -> Transcript {
53 let text = String::from_utf8_lossy(bytes);
54 let mut out = Transcript::default();
55
56 for line in text.lines() {
57 let line = line.trim();
58 if line.is_empty() {
59 continue;
60 }
61 let Ok(entry) = serde_json::from_str::<Entry>(line) else {
62 continue;
63 };
64
65 if let Some(version) = entry.version {
66 out.agent_version = Some(version);
67 }
68
69 let Some(message) = entry.message else {
70 continue;
71 };
72
73 if let Some(model_id) = message.model {
74 out.model = Some(Model {
75 provider: provider_of(&model_id),
76 id: model_id,
77 });
78 }
79
80 if let Some(usage) = message.usage {
81 out.usage.input_tokens = out.usage.input_tokens.saturating_add(usage.input_tokens);
82 out.usage.output_tokens = out.usage.output_tokens.saturating_add(usage.output_tokens);
83 }
84
85 if entry.kind.as_deref() == Some("assistant") {
86 let text = message
87 .content
88 .as_ref()
89 .map(content_text)
90 .unwrap_or_default();
91 if !text.is_empty() {
92 out.assistant_texts.push(text);
93 }
94 }
95 }
96
97 out
98}

100/// Rät den Anbieter aus der Modell-ID. Bewusst grob — die ID ist die Wahrheit,
101/// der Anbieter nur eine Bequemlichkeit für den Reader.
102pub(crate) fn provider_of(model_id: &str) -> String {
103 let id = model_id.to_ascii_lowercase();
104 if id.starts_with("claude") {
105 "anthropic"
106 } else if id.starts_with("gpt") || is_openai_o_series(&id) {
107 "openai"
108 } else if id.starts_with("gemini") {
109 "google"
110 } else {
111 "unknown"
112 }
113 .to_string()
114}

116/// OpenAIs o-Serie: `o1`, `o3`, `o4-mini`, … — ein `o`, gefolgt von einer
117/// Ziffer. Bewusst eng, damit `opus` nicht fälschlich als OpenAI gilt.
118fn is_openai_o_series(id: &str) -> bool {
119 let mut chars = id.chars();
120 chars.next() == Some('o') && chars.next().is_some_and(|c| c.is_ascii_digit())
121}

127#[derive(Debug, Deserialize)]
128struct Entry {
129 #[serde(rename = "type")]
130 kind: Option<String>,
131 version: Option<String>,
132 message: Option<Message>,
133}

135#[derive(Debug, Deserialize)]
136struct Message {
137 model: Option<String>,
138 usage: Option<UsageWire>,
139 /// Roh als [`serde_json::Value`] gehalten statt in ein untagged Enum
140 /// gezwängt: Der Inhalt ist mal ein String (User), mal eine Block-Liste
141 /// (Assistant), und ein unerwarteter dritter Fall darf die ganze Zeile nicht
142 /// verwerfen. Die Deutung übernimmt [`content_text`].
143 content: Option<serde_json::Value>,
144}
145
146#[derive(Debug, Deserialize)]
147struct UsageWire {
148 #[serde(default)]
149 input_tokens: u64,
150 #[serde(default)]
151 output_tokens: u64,
152}
153
154/// Der Antworttext eines Nachrichteninhalts.
155///
156/// Ein String ist der User-Fall. Eine Liste ist der Assistant-Fall: nur die
157/// `text`-Blöcke tragen Antworttext; `thinking` und `tool_use` werden
158/// übergangen. Alles Übrige ergibt leeren Text — nie einen Fehler.
159fn content_text(content: &serde_json::Value) -> String {
160 match content {
161 serde_json::Value::String(s) => s.clone(),
162 serde_json::Value::Array(blocks) => blocks
163 .iter()
164 .filter(|b| b.get("type").and_then(|t| t.as_str()) == Some("text"))
165 .filter_map(|b| b.get("text").and_then(|t| t.as_str()))
166 .collect::<Vec<_>>()
167 .join("\n"),
168 _ => String::new(),
169 }
170}

Das Verhalten soll an der Stelle absolut gleich sein.
