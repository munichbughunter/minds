1//! Der Index: alles, was die Seite braucht, einmal aus Git und Store gezogen.
2//!
3//! Der Reader ist **zustandslos** (Architektur-Prinzip 6): Er hält keine
4//! Datenbank, sondern baut bei jedem Lauf ein Bild aus zwei Quellen — den
5//! Sessions im Store und den Trailern in der Historie. Dieses Modul ist dieses
6//! Bild.
7//!
8//! ```text
9//! Store ──list/get──► SessionId → Session
10//! Repo ──revwalk───► Commit → [SessionId]
11//! ```
12//!
13//! # Zwei Richtungen, eine Wahrheit
14//!
15//! Die verbindliche Verknüpfung ist der **Trailer** (Commit → Session); genau
16//! die wird hier gesammelt. Der Store liefert dazu die Nutzlast. Was der Index
17//! *nicht* tut, ist raten: Eine Session ohne Trailer taucht in `sessions` auf,
18//! aber unter keinem Commit — sie ist erfasst, aber (noch) nicht mit Code
19//! verbunden. Das ist ein legitimer Zustand und kein Fehler.
20//!
21//! # Eine kaputte Session bringt nicht die Seite zu Fall
22//!
23//! Lässt sich eine Session nicht auflösen (Inhalt passt nicht zum Hash, kaputtes
24//! JSON), wird sie übersprungen und **gezählt** statt verschwiegen — siehe
25//! [`Index::unreadable`]. Der Reader ist ein Leser; er darf an einem faulen
26//! Eintrag nicht sterben, aber er darf ihn auch nicht unterschlagen. `minds
27//! fsck` ist das Werkzeug, um dem nachzugehen.
28

37/// Das gesammelte Bild aus Store und Historie.
38///
39/// Zwei Quellen verknüpfen Commits mit Sessions: die **Trailer** (verbindlich)
40/// und der **Store-Index** (heuristisch, für importierte Sessions). Der Index
41/// führt beide zusammen und merkt sich in [`Index::observed`], welche Sessions
42/// über mindestens einen Trailer belegt sind — der Rest ist „vermutet" und wird
43/// im Reader als solcher gezeigt.
44#[derive(Debug, Default, Clone)]
45pub struct Index {
46 sessions: BTreeMap<SessionId, Session>,
47 commits: BTreeMap<CommitId, Vec<SessionId>>,
48 observed: BTreeSet<SessionId>,
49 unreadable: usize,
50}
51
52impl Index {
53 /// Zieht Sessions, Trailer und den Store-Index aus Store und Repository.
54 pub fn build(repo: &Repo, store: &dyn ContextStore) -> Result<Self> {
55 let mut sessions = BTreeMap::new();
56 let mut unreadable = 0usize;
57
58 for id in store.list()? {
59 match store.get(id) {
60 // Sessions ohne erfasste Absicht tragen nichts bei und werden
61 // gar nicht erst aufgenommen — dann verschwinden sie überall:
62 // Übersicht, Datei-Panels und die Zeilen-Zuordnung. Das ist der
63 // frühere „(kein Prompt erfasst)"-Ballast, an einer Stelle
64 // ausgesiebt.
65 Ok(Some(session)) if session.intent.request.trim().is_empty() => {}
66 Ok(Some(session)) => {
67 sessions.insert(id, session);
68 }
69 // Im Store gelistet, aber nicht auflösbar — zählen, nicht sterben.
70 Ok(None) | Err(_) => unreadable += 1,
71 }
72 }
73
74 let mut commits: BTreeMap<CommitId, Vec<SessionId>> = BTreeMap::new();
75 let mut observed: BTreeSet<SessionId> = BTreeSet::new();
76
77 // 1. Die Trailer — die verbindliche Richtung. Nur Kanten zu Sessions,
78 // die wir behalten haben (keine ausgesiebten).
79 if let Some(head) = repo.head()?.commit() {
80 for commit in repo.revwalk(head)? {
81 let commit = commit?;
82 for id in repo.session_ids_of(commit)? {
83 if sessions.contains_key(&id) {
84 observed.insert(id);
85 push_unique(commits.entry(commit).or_default(), id);
86 }
87 }
88 }
89 }
90
91 // 2. Der Store-Index — die vermuteten Kanten (z. B. importiert). Was
92 // schon über einen Trailer da ist, wird nicht doppelt geführt.
93 for (hex, links) in store.index()?.iter() {
94 let Ok(commit) = hex.parse::<CommitId>() else {
95 continue;
96 };
97 for link in links {
98 if sessions.contains_key(&link.session) {
99 push_unique(commits.entry(commit).or_default(), link.session);
100 }
101 }
102 }
103
104 Ok(Self {
105 sessions,
106 commits,
107 observed,
108 unreadable,
109 })
110 }
111
112 /// Baut einen Index aus fertigen Teilen — für Tests und für Aufrufer, die
113 /// ihre Daten schon haben. Alle Verknüpfungen gelten dabei als beobachtet.
114 ///
115 /// Auch hier fliegen absichtslose Sessions und Kanten auf sie raus, damit
116 /// der Test denselben Vertrag prüft wie [`Index::build`].
117 pub fn from_parts(
118 sessions: BTreeMap<SessionId, Session>,
119 commits: BTreeMap<CommitId, Vec<SessionId>>,
120 ) -> Self {
121 let sessions: BTreeMap<SessionId, Session> = sessions
122 .into_iter()
123 .filter(|(_, s)| !s.intent.request.trim().is_empty())
124 .collect();
125 let commits: BTreeMap<CommitId, Vec<SessionId>> = commits
126 .into_iter()
127 .map(|(commit, ids)| {
128 (
129 commit,
130 ids.into_iter()
131 .filter(|id| sessions.contains_key(id))
132 .collect::<Vec<_>>(),
133 )
134 })
135 .filter(|(_, ids)| !ids.is_empty())
136 .collect();
137 let observed = commits.values().flatten().copied().collect();
138 Self {
139 sessions,
140 commits,
141 observed,
142 unreadable: 0,
143 }
144 }

146 /// `true`, wenn diese Session über mindestens einen Trailer belegt ist (im

148 pub fn is_observed(&self, id: SessionId) -> bool {

188/// Fügt `id` an, wenn sie nicht schon in `ids` steht.
189fn push_unique(ids: &mut Vec<SessionId>, id: SessionId) {
190 if !ids.contains(&id) {
