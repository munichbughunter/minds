1//! Die Seite bauen — der einzige Teil des Readers, der Dateien schreibt.
2//!
3//! ```text
4//! Index bauen ──► je Datei: Blob + Blame + join ──► HTML schreiben
5//! ```
6//!
7//! Ergebnis ist ein Verzeichnis mit `index.html` und je einer Seite pro Datei,
8//! die erfassten Kontext trägt. Nichts darin verweist nach außen, also lässt es
9//! sich per `file://` öffnen, hinter jede Firewall stellen und in ein Air-Gap
10//! kopieren.
11//!
12//! # Was gerendert wird — und was nicht
13//!
14//! Geschrieben wird eine Seite nur für Dateien mit **mindestens einer
15//! zugeordneten Zeile**. Eine Datei, an der nie ein erfasster Agent gearbeitet
16//! hat, hätte nichts zu zeigen; sie wegzulassen hält die Ausgabe bei dem, was
17//! belegt ist.
18//!
19//! # Kosten, ehrlich benannt
20//!
21//! Für die Zuordnung wird **jede Datei im Baum von HEAD geblamed** — ein Blame
22//! pro Datei. Das ist der korrekte, einfache Weg und für Repositories üblicher
23//! Größe schnell genug; auf sehr großen Bäumen ist es der teuerste Teil des
24//! Laufs. Ihn zu verengen (nur Dateien, die von getrailerten Commits berührt
25//! wurden) braucht eine Diff-Schnittstelle in `minds-git`, die es noch nicht
26//! gibt — deshalb steht hier die einfache Variante und keine Heuristik, die
27//! stillschweigend etwas ausließe.
28//!
29//! Dateien, die kein UTF-8 sind (Bilder, Binaries), werden übersprungen; ein
30//! Blame, der scheitert, ebenfalls. Beides wird **gezählt** und im Ergebnis
31//! ausgewiesen statt verschwiegen.

59/// Baut die statische Seite nach `out`.
60pub fn render(repo: &Repo, store: &dyn ContextStore, out: &Path) -> Result<Site> {
61 let index = Index::build(repo, store)?;
62 let head = repo.head()?.commit().ok_or(ReaderError::UnbornHead)?;
63
64 std::fs::create_dir_all(out)
65 .map_err(|e| ReaderError::io("Ausgabeverzeichnis anlegen", out, e))?;
66
67 let mut links: Vec<FileLink> = Vec::new();
68 let mut used: BTreeSet<String> = BTreeSet::new();
69 let mut skipped = 0usize;
70 // Session → die erste Datei-Seite, die sie zeigt. Damit wird eine
71 // Übersichts-Karte klickbar und führt auf genau die Datei-Ansicht, in der
72 // diese Session hängt (dort öffnet ein Anker ihr Panel).
73 let mut session_page: BTreeMap<SessionId, String> = BTreeMap::new();
74
75 // Trägt kein Commit einen Trailer, kann auch keine Zeile zugeordnet sein —
76 // dann ist jeder Blame verschwendet. Das ist der Zustand eines Repos, in
77 // dem Minds gerade erst eingerichtet wurde, also der häufigste erste Lauf.
78 let candidates = if index.attributed_commits() == 0 {
79 Vec::new()
80 } else {
81 repo.list_blobs_at("HEAD")?
82 };
83
84 for path in candidates {
85 let Some(bytes) = repo.read_blob_at("HEAD", &path)? else {
86 continue;
87 };
88 // Binärdateien haben keine Zeilen, die man anklicken könnte.
89 let Ok(content) = String::from_utf8(bytes) else {
90 skipped += 1;
91 continue;
92 };
93 let Ok(blame) = repo.blame().blame_file(head, &path) else {
94 skipped += 1;
95 continue;
96 };
97
98 let view = FileView::join(&path, &content, &blame, &index);
99 if !view.is_attributed() {
100 continue;
101 }
102
103 let href = unique_slug(&path, &mut used);
104 for id in view.sessions() {
105 session_page.entry(id).or_insert_with(|| href.clone());
106 }
107 write(&out.join(&href), &html::file_page(&view, &index))?;
108 links.push(FileLink {
109 attributed: view.attributed_lines(),
110 total: view.lines.len(),
111 path,
112 href,
113 });
114 }
115
116 write(
117 &out.join("index.html"),
118 &html::index_page(&index, &links, &session_page),
119 )?;
120
121 Ok(Site {
122 out: out.to_path_buf(),
123 files: links.len(),
124 sessions: index.len(),
125 skipped,
126 })
127}

129/// Ein Dateiname, der in diesem Lauf noch nicht vergeben ist.
130///
131/// [`html::slug`] ist nicht injektiv (`a/b` und `a-b` fallen zusammen). Statt
132/// eines Hashes, der den Namen unlesbar machte, wird bei Kollision
133/// durchnummeriert — deterministisch, weil die Dateien in der sortierten
134/// Reihenfolge von `list_blobs_at` verarbeitet werden.
135fn unique_slug(path: &str, used: &mut BTreeSet<String>) -> String {
136 let base = html::slug(path);
137 if used.insert(base.clone()) {
138 return base;
139 }
140
141 let stem = base.strip_suffix(".html").unwrap_or(&base).to_string();
142 for n in 2u32.. {
143 let candidate = format!("{stem}-{n}.html");
144 if used.insert(candidate.clone()) {
145 return candidate;
146 }
147 }
148 unreachable!("u32 reicht für Namenskollisionen")
149}

151fn write(path: &Path, contents: &str) -> Result<()> {
152 std::fs::write(path, contents).map_err(|e| ReaderError::io("Seite schreiben", path, e))
153}
