1//! Das lokale Journal: eine Datei pro Hook-Event, 0600, außerhalb von Git.
2//!
3//! Ein Agent-Hook ist ein Prozessstart mitten in der Arbeitsschleife des
4//! Nutzers. Er darf drei Dinge tun — stdin lesen, eine Datei schreiben, mit 0
5//! enden — und sonst nichts. Kein Git, keine Redaction, kein Parsen des
6//! Transkripts. Alles Teure passiert später, beim Checkpoint. Dieses Modul ist
7//! der „eine Datei schreiben"-Teil.
8//!
9//! # Warum das Journal kein Git-Ref ist
10//!
11//! Das Journal enthält **Rohdaten**: ungefilterte Tool-Ausgaben, Prompts,
12//! möglicherweise Zugangsdaten. Genau hier hat der Stand der Technik seine
13//! dokumentierte Schwachstelle — temporäre Shadow-Branches, die unredigierte
14//! Daten enthalten können und „nicht gepusht werden sollten".
15//!
16//! „Sollte nicht" ist keine Garantie. Deshalb liegt das Journal unter
17//! `<git-dir>/minds/journal/`: außerhalb des Worktrees, kein Ref, kein Objekt,
18//! nicht im Index. Es kann nicht versehentlich committet und nicht
19//! versehentlich gepusht werden, weil Git es schlicht nicht kennt. Rechte sind
20//! 0700/0600. Nach erfolgreicher Übernahme in den Store wird es gelöscht.
21//!
22//! Das ist kein besserer Vorsatz, sondern eine andere Bauform: Die Zusage ist
23//! strukturell, nicht disziplinarisch — dieselbe Linie wie beim
24//! `RedactionAudit`, der keine Zeichenkette aus dem Eingabetext enthält.
25//!
26//! # Layout
27//!
28//! ```text
29//! <git-dir>/minds/journal/
30//! └─ claude-code/ # Agentname
31//! └─ 31f3f224-f440-41ac-.../ # local_id der Agent-Session
32//! ├─ 0000000000.json # seq 0
33//! ├─ 0000000001.json
34//! └─ .next # Hinweis, unverbindlich
35//! ```
36//!
37//! # Die Sequenznummer, und warum sie ohne Sperre auskommt
38//!
39//! Mehrere Agents parallel heißt mehrere gleichzeitige Hook-Prozesse. Ein
40//! gemeinsames Append-Log bräuchte eine Sperre auf dem heißen Pfad; eine Sperre
41//! bedeutet Wartezeit im Agenten des Nutzers und im Absturzfall eine
42//! verwaiste Sperrdatei.
43//!
44//! Stattdessen ist die **Dateierstellung selbst die Sperre**:
45//! [`File::create_new`] ist atomar. Wer `0000000042.json` anlegen kann, besitzt
46//! Sequenznummer 42; wer scheitert, zählt hoch und versucht es erneut. Zwei
47//! Prozesse können sich nie dieselbe Nummer teilen, ohne dass jemals ein Lock
48//! gehalten wird. Die Datei `.next` ist nur ein Startwert für die Suche und
49//! darf beliebig falsch sein — sie spart den `read_dir`, sie garantiert nichts.
50//!
51//! Die Nummer ist dabei kein Ordnungs-Luxus, sondern der einzige Weg, **Lücken**
52//! zu erkennen: Der Hook ist fail-open (er darf die Sitzung des Nutzers nie
53//! abbrechen), also *kann* ein Event fehlen. Ein Sprung in der Folge macht das
54//! sichtbar. Ehrlich lückenhaft schlägt still vollständig.
55//!
56//! # Schreiben in zwei Schritten
57//!
58//! `create_new` liefert eine **leere** Datei — ein Leser, der genau in diesem
59//! Moment schaut, sähe kein halbes JSON, sondern null Bytes. Der Inhalt geht
60//! deshalb nach `<seq>.json.tmp`, wird gefsynct und dann über die Reservierung
61//! **umbenannt**; `rename` ist atomar. Übrig gebliebene leere Dateien oder
62//! `.tmp`-Reste sind damit eindeutig als abgestürzter Schreibvorgang lesbar und
63//! nicht als gültiges Event — [`Journal::read`] meldet sie als
64//! [`Damaged`](ReadOutcome::Damaged), statt sie zu verschweigen.
65

101impl SessionKey {
102 /// Prüft beide Bestandteile und baut den Schlüssel.
103 ///
104 /// # Warum das eine Sicherheitsgrenze ist
105 ///
106 /// `local_id` kommt aus dem JSON, das der Agent auf stdin schickt — also
107 /// aus einer Quelle, die wir nicht kontrollieren. Ungeprüft in einen Pfad
108 /// eingesetzt, wäre `../../hooks/pre-commit` ein Schreibzugriff auf das
109 /// Git-Verzeichnis. Erlaubt sind deshalb nur `[A-Za-z0-9._-]`, nicht leer,
110 /// nicht `.` oder `..`, höchstens 128 Zeichen. UUIDs und Agentnamen passen
111 /// da bequem hinein; alles andere wird abgelehnt statt zurechtgebogen.
112 pub fn new(agent: impl Into<String>, local_id: impl Into<String>) -> Result<Self> {
113 let agent = agent.into();
114 let local_id = local_id.into();
115 check_component(&agent, "agent")?;
116 check_component(&local_id, "local_id")?;
117 Ok(Self { agent, local_id })
118 }

120 pub fn agent(&self) -> &str {
121 &self.agent
122 }
123
124 pub fn local_id(&self) -> &str {
125 &self.local_id
126 }
127}

129fn check_component(value: &str, field: &'static str) -> Result<()> {
130 let ok = !value.is_empty()
131 && value.len() <= 128
132 && value != "."
133 && value != ".."
134 && value
135 .bytes()
136 .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-'));
137
138 if ok {
139 Ok(())
140 } else {
141 Err(CaptureError::UnsafeKey {
142 field,
143 value: value.chars().take(64).collect(),
144 })
145 }
146}

152/// Ein Hook-Event, wie es im Journal liegt.
153///
154/// Das Format ist **unseres**, nicht das des Agenten: Ein normalisierter
155/// Umschlag plus der unveränderte Original-Payload. Der Umschlag macht das
156/// Journal agent-unabhängig lesbar, der Payload bleibt Beweismittel — was wir
157/// heute nicht zu deuten wissen, ist morgen noch da.
158///
159/// Dieses Format wird **nie gehasht** und ist nicht Teil der
160/// Content-Adressierung. Die Beschränkungen der kanonischen Form (nur
161/// Ganzzahlen unter 2^53) gelten hier deshalb nicht.
162#[derive(Debug, Clone, Serialize, Deserialize)]
163pub struct JournalEvent {
164 /// Laufende Nummer innerhalb dieser Session, lückenlos ab 0 — siehe
165 /// Modul-Doku.
166 pub seq: u64,
167
168 /// Zeitpunkt der Aufzeichnung, RFC 3339 in UTC. Von *uns* gesetzt, nicht
169 /// vom Agenten: Der Hook ist dabei, wenn es passiert, und ist damit die
170 /// bessere Quelle als ein Feld, das nicht jeder Agent mitschickt.
171 pub at: String,
172
173 /// Dieselbe Ablesung in Nanosekunden seit Epoch — der Sortierschlüssel,
174 /// wenn Events *verschiedener* Agents zusammengeführt werden.
175 pub at_nanos: u64,
176
177 /// Auf unser Vokabular normalisierte Art des Events.
178 pub kind: EventKind,
179
180 /// Der Originalname des Events beim Agenten (`PostToolUse`, …). Bleibt
181 /// erhalten, auch wenn [`EventKind`] ihn auf [`EventKind::Other`] abbildet.
182 pub raw_kind: String,
183
184 /// Arbeitsverzeichnis laut Hook-Payload.
185 #[serde(default, skip_serializing_if = "Option::is_none")]
186 pub cwd: Option<String>,
187
188 /// Pfad zum Transkript des Agenten, falls mitgeliefert.
189 ///
190 /// Das ist die Klammer zwischen beiden Welten: Der Hook liefert Zeitpunkt,
191 /// Reihenfolge und Kausalität, das Transkript den reichen Inhalt (Volltext,
192 /// Thinking, Token-Zähler), der im Hook-Payload nicht steht. Beide Hälften
193 /// werden erst beim Checkpoint zusammengeführt.
194 #[serde(default, skip_serializing_if = "Option::is_none")]
195 pub transcript_path: Option<String>,
196
197 /// Der unveränderte Payload des Agenten.
198 ///
199 /// [`RawValue`] statt [`serde_json::Value`], damit die Bytes exakt so
200 /// erhalten bleiben, wie sie ankamen — kein Umsortieren von Schlüsseln,
201 /// keine Zahlen, die durch einen Parse/Serialize-Zyklus ihre Schreibweise
202 /// ändern. Beweismittel werden nicht umformatiert.
203 pub payload: Box<RawValue>,
204}

219///
220/// Klein gehalten: Die Lebenszyklen der gängigen Agents lassen sich auf

233 SubagentStart,

235 SessionEnd,

237}

249impl Journal {
250 /// Öffnet das Journal unterhalb eines bekannten Git-Verzeichnisses.
251 ///
252 /// Legt nichts an — das passiert erst beim ersten [`append`](Self::append).
253 /// Ein Hook, der nichts zu schreiben hat, hinterlässt keine Spur.
254 pub fn open(git_dir: &Path) -> Self {
255 Self {
256 root: git_dir.join(JOURNAL_DIR),
257 }
258 }
259
260 /// Sucht von `start` aufwärts ein `.git` und öffnet das Journal darin.
261 ///
262 /// Bewusst eine eigene, dumme Suche statt `minds-git`: Der Hook soll auf
263 /// dem heißen Pfad kein Repository öffnen, keine Refs auflösen und keine
264 /// Konfiguration lesen. Er braucht nur ein Verzeichnis.
265 ///
266 /// `.git` als **Datei** (verlinkte Worktrees, Submodule) wird gelesen und
267 /// dem `gitdir:`-Verweis gefolgt. Ohne das hätte jeder Worktree stillschweigend
268 /// kein Journal.
269 pub fn discover(start: &Path) -> Result<Self> {
270 let start = start
271 .canonicalize()
272 .map_err(|e| CaptureError::io("Arbeitsverzeichnis auflösen", start, e))?;
273
274 for dir in start.ancestors() {
275 let candidate = dir.join(".git");
276 match fs::metadata(&candidate) {
277 Ok(m) if m.is_dir() => return Ok(Self::open(&candidate)),
278 Ok(m) if m.is_file() => {
279 let text = fs::read_to_string(&candidate)
280 .map_err(|e| CaptureError::io("gitdir-Datei lesen", &candidate, e))?;
281 let target = text
282 .lines()
283 .find_map(|l| l.strip_prefix("gitdir:"))
284 .ok_or_else(|| CaptureError::NoRepository {
285 start: start.clone(),
286 })?
287 .trim();
288 let target = dir.join(target);
289 return Ok(Self::open(&target));
290 }
291 _ => continue,
292 }
293 }
294
295 Err(CaptureError::NoRepository { start })
296 }

300 &self.root

303 /// Hängt ein Event an und gibt es mit vergebener Sequenznummer zurück.
304 ///
305 /// Das ist die einzige Operation auf dem heißen Pfad. Sie macht: bis zu
306 /// zwei `create`-Aufrufe, ein `write`, ein `fsync`, ein `rename`. Kein
307 /// Netz, kein Git, keine Sperre.
308 pub fn append(&self, key: &SessionKey, event: NewEvent) -> Result<JournalEvent> {
309 let dir = self.session_dir(key);
310 create_dir_private(&dir)?;
311
312 let (seq, claim) = self.reserve(&dir)?;
313
314 let event = JournalEvent {
315 seq,
316 at: event.at,
317 at_nanos: event.at_nanos,
318 kind: event.kind,
319 raw_kind: event.raw_kind,
320 cwd: event.cwd,
321 transcript_path: event.transcript_path,
322 payload: event.payload,
323 };
324
325 let tmp = dir.join(format!("{seq:010}.json.tmp"));
326 write_private(&tmp, &serde_json::to_vec(&event)?)?;
327 fs::rename(&tmp, &claim).map_err(|e| CaptureError::io("Event umbenennen", &claim, e))?;
328
329 // Unverbindlicher Startwert fuer den naechsten Aufruf. Schlaegt das
330 // fehl, kostet es genau einen `read_dir` — kein Grund, das Event
331 // zurueckzurollen.
332 let _ = fs::write(dir.join(HINT_FILE), (seq + 1).to_string());
333
334 Ok(event)
335 }

337 /// Reserviert die nächste freie Sequenznummer, indem sie ihre Datei anlegt.
338 fn reserve(&self, dir: &Path) -> Result<(u64, PathBuf)> {
339 let mut seq = read_hint(dir).unwrap_or_else(|| scan_next_seq(dir));
340
341 for _ in 0..MAX_SEQ_PROBES {
342 let path = dir.join(format!("{seq:010}.json"));
343 match File::create_new(&path) {
344 Ok(_) => return Ok((seq, path)),
345 Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists => {
346 seq += 1;
347 }
348 Err(e) => return Err(CaptureError::io("Event anlegen", &path, e)),
349 }
350 }
351
352 Err(CaptureError::SeqExhausted {
353 dir: dir.to_path_buf(),
354 probes: MAX_SEQ_PROBES,
355 })
356 }

358 /// Alle Sessions, für die Events vorliegen.
359 pub fn sessions(&self) -> Result<Vec<SessionKey>> {
360 let mut out = Vec::new();
361 let agents = match fs::read_dir(&self.root) {
362 Ok(it) => it,
363 Err(e) if e.kind() == std::io::ErrorKind::NotFound => return Ok(out),
364 Err(e) => return Err(CaptureError::io("Journal lesen", &self.root, e)),
365 };
366
367 for agent in agents.flatten() {
368 if !agent.file_type().map(|t| t.is_dir()).unwrap_or(false) {
369 continue;
370 }
371 let agent_name = agent.file_name().to_string_lossy().into_owned();
372 let Ok(sessions) = fs::read_dir(agent.path()) else {
373 continue;
374 };
375 for session in sessions.flatten() {
376 if !session.file_type().map(|t| t.is_dir()).unwrap_or(false) {
377 continue;
378 }
379 let local_id = session.file_name().to_string_lossy().into_owned();
380 if let Ok(key) = SessionKey::new(agent_name.clone(), local_id) {
381 out.push(key);
382 }
383 }
384 }
385
386 out.sort();
387 Ok(out)
388 }

390 /// Liest alle Events einer Session, nach Sequenznummer sortiert.
391 ///
392 /// Meldet Lücken und beschädigte Dateien mit, statt sie zu verschweigen —
393 /// siehe [`ReadOutcome`].
394 pub fn read(&self, key: &SessionKey) -> Result<ReadOutcome> {
395 let dir = self.session_dir(key);
396 let mut events = Vec::new();
397 let mut damaged = Vec::new();
398
399 let entries = match fs::read_dir(&dir) {
400 Ok(it) => it,
401 Err(e) if e.kind() == std::io::ErrorKind::NotFound => {
402 return Ok(ReadOutcome::default());
403 }
404 Err(e) => return Err(CaptureError::io("Session lesen", &dir, e)),
405 };
406
407 for entry in entries.flatten() {
408 let path = entry.path();
409 let Some(name) = path.file_name().and_then(|n| n.to_str()) else {
410 continue;
411 };
412 if !name.ends_with(".json") {
413 // `.tmp`-Reste und `.next` sind kein Event.
414 if name.ends_with(".json.tmp") {
415 damaged.push(path);
416 }
417 continue;
418 }
419
420 match fs::read(&path) {
421 Ok(bytes) if bytes.is_empty() => damaged.push(path),
422 Ok(bytes) => match serde_json::from_slice::<JournalEvent>(&bytes) {
423 Ok(ev) => events.push(ev),
424 Err(_) => damaged.push(path),
425 },
426 Err(_) => damaged.push(path),
427 }
428 }
429
430 events.sort_by_key(|e| e.seq);
431 let gaps = gaps(&events);
432
433 Ok(ReadOutcome {
434 events,
435 gaps,
436 damaged,
437 })
438 }
439
440 /// Löscht alle Events einer Session — der Schritt *nach* erfolgreicher
441 /// Übernahme in den Store.
442 ///
443 /// Bis dahin bleibt das Journal liegen. Ein Absturz zwischen Capture und
444 /// Store darf keine Rohdaten verlieren; er darf sie nur nicht behalten,
445 /// wenn sie sicher angekommen sind.
446 pub fn discard(&self, key: &SessionKey) -> Result<()> {
447 let dir = self.session_dir(key);
448 match fs::remove_dir_all(&dir) {
449 Ok(()) => Ok(()),
450 Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(()),
451 Err(e) => Err(CaptureError::io("Session verwerfen", &dir, e)),
452 }
453 }
454
455 fn session_dir(&self, key: &SessionKey) -> PathBuf {
456 self.root.join(&key.agent).join(&key.local_id)
457 }
458}

460/// Ergebnis von [`Journal::read`] — inklusive dem, was fehlt.
461#[derive(Debug, Default)]
462pub struct ReadOutcome {
463 /// Die gelesenen Events, nach `seq` sortiert.
464 pub events: Vec<JournalEvent>,
465
466 /// Sequenznummern, die zwischen der kleinsten und der größten vorhandenen
467 /// fehlen. Nicht leer heißt: Der Hook ist mindestens einmal fail-open
468 /// gelaufen. `minds fsck` macht das sichtbar.
469 pub gaps: Vec<u64>,
470
471 /// Dateien, die kein gültiges Event enthielten (leere Reservierung,
472 /// `.tmp`-Rest, kaputtes JSON) — ein abgestürzter Schreibvorgang.
473 pub damaged: Vec<PathBuf>,
474}
475
476impl ReadOutcome {
477 /// `true`, wenn nichts fehlt und nichts beschädigt ist.
478 pub fn is_complete(&self) -> bool {
479 self.gaps.is_empty() && self.damaged.is_empty()
480 }
481}
482
483/// Fehlende Sequenznummern zwischen der ersten und der letzten vorhandenen.
484///
485/// Bewusst **nicht** ab 0 gezählt: Wenn `minds capture` einen Teil des Journals
486/// bereits übernommen und gelöscht hat, beginnt der Rest legitim bei einer
487/// höheren Nummer. Eine Lücke ist etwas Fehlendes *zwischen* Vorhandenem.
488fn gaps(sorted: &[JournalEvent]) -> Vec<u64> {
489 let (Some(first), Some(last)) = (sorted.first(), sorted.last()) else {
490 return Vec::new();
491 };
492 let mut missing = Vec::new();
493 let mut expect = first.seq;
494 for ev in sorted {
495 while expect < ev.seq {
496 missing.push(expect);
497 expect += 1;
498 }
499 expect = ev.seq + 1;
500 }
501 debug_assert!(expect == last.seq + 1);
502 missing
503}

517fn scan_next_seq(dir: &Path) -> u64 {
518 let Ok(entries) = fs::read_dir(dir) else {
519 return 0;
520 };
521 entries
522 .flatten()
523 .filter_map(|e| {
524 let name = e.file_name();
525 let name = name.to_str()?;
526 name.strip_suffix(".json")?.parse::<u64>().ok()
527 })
528 .max()
529 .map(|m| m + 1)
530 .unwrap_or(0)
531}

533fn create_dir_private(dir: &Path) -> Result<()> {
534 fs::create_dir_all(dir).map_err(|e| CaptureError::io("Journal-Verzeichnis anlegen", dir, e))?;
535 #[cfg(unix)]
536 {
537 use std::os::unix::fs::PermissionsExt;
538 // Nur der Eigentuemer. Das Journal enthaelt Rohdaten.
539 let _ = fs::set_permissions(dir, fs::Permissions::from_mode(0o700));
540 }
541 Ok(())
542}
543
544fn write_private(path: &Path, bytes: &[u8]) -> Result<()> {
545 let mut opts = OpenOptions::new();
546 opts.write(true).create(true).truncate(true);
547 #[cfg(unix)]
548 {
549 use std::os::unix::fs::OpenOptionsExt;
550 opts.mode(0o600);
551 }
552 let mut f = opts
553 .open(path)
554 .map_err(|e| CaptureError::io("Event schreiben", path, e))?;
555 f.write_all(bytes)
556 .map_err(|e| CaptureError::io("Event schreiben", path, e))?;
557 // Vor dem `rename` synchronisieren: Sonst kann ein Absturz eine sichtbare,
558 // aber leere Datei hinterlassen — also ein Event, das es nie gab.
559 f.sync_all()
560 .map_err(|e| CaptureError::io("Event synchronisieren", path, e))?;
561 Ok(())
562}
