1//! End-to-End-Fixtures: roher Hook-Payload rein, [`Session`] raus.
2//!
3//! Diese Tests fahren den *ganzen* heißen und kalten Pfad, den `minds` in echt
4//! fährt — nur die Uhr wird hereingereicht, damit die Ergebnisse Byte für Byte
5//! reproduzierbar sind:
6//!
7//! ```text
8//! payload ──► hook_event::parse ──► secretwall::guard ──► Journal::append
9//! │
10//! adapter::build ◄──────────────────────────────┘
11//! ```
12//!
13//! Drei Szenarien aus der Anforderung: eine einzelne Session, ein Sub-Agent, und
14//! zwei Agents, die parallel in *dasselbe* Journal schreiben — der Fall, den ein
15//! Transkript-Parser prinzipiell nicht sehen könnte.

22/// Ein fester Zeitstempel, abgeleitet aus einer Sequenz — so ist jeder Testlauf
23/// derselbe, und `at`/`at_nanos` bleiben konsistent.
24fn at(n: u64) -> (String, u64) {
25 let nanos = 1_784_797_924_000_000_000 + n * 1_000_000_000;
26 (clock::rfc3339_from_nanos(nanos), nanos)
27}
28
29/// Genau das, was `minds hook` tut: parsen, die Secretfile-Mauer anwenden,
30/// anhängen. Der Agentname kommt aus der Registrierung, nie aus dem Payload.
31fn feed(journal: &Journal, agent: &str, event: Option<&str>, payload: &str, seq: u64) {
32 let mut parsed = hook_event::parse(payload.as_bytes().to_vec(), agent, event, at(seq)).unwrap();
33 secretwall::guard(&mut parsed.event);
34 journal.append(&parsed.key, parsed.event).unwrap();
35}

158#[test]
159fn subagent_edges_are_observed_both_ways() {
160 let tmp = tempfile::tempdir().unwrap();
161 let journal = journal_in(tmp.path());
162
163 // Elternteil: sieht seinen SubagentEnd, der das Kind nennt.
164 feed(
165 &journal,
166 "claude-code",
167 None,
168 r#"{"session_id":"s-parent","hook_event_name":"UserPromptSubmit","prompt":"delegiere das Review"}"#,
169 0,
170 );
171 feed(
172 &journal,
173 "claude-code",
174 None,
175 r#"{"session_id":"s-parent","hook_event_name":"SubagentStop","subagent_session_id":"s-child","subagent_agent":"claude-code"}"#,
176 1,
177 );
178
179 // Kind: sieht seinen eigenen Start als Sub-Agent, nennt den Elternteil.
180 feed(
181 &journal,
182 "claude-code",
183 None,
184 r#"{"session_id":"s-child","hook_event_name":"SessionStart","source":"subagent","parent_session_id":"s-parent"}"#,
185 2,
186 );
187 feed(
188 &journal,
189 "claude-code",
190 None,
191 r#"{"session_id":"s-child","hook_event_name":"Stop"}"#,
192 3,
193 );
194
195 let sessions = adapter::build(&journal).unwrap();
196 assert_eq!(sessions.len(), 2);
197
198 let parent = find(&sessions, "s-parent");
199 let child = find(&sessions, "s-child");
200
201 // Elternteil → Kind, Spawned, beobachtet.
202 assert_eq!(parent.edges.len(), 1);
203 assert_eq!(parent.edges[0].kind, EdgeKind::Spawned);
204 assert_eq!(parent.edges[0].evidence, Evidence::Observed);
205 assert_eq!(
206 parent.edges[0].to,
207 Endpoint::Session {
208 agent: "claude-code".into(),
209 local_id: "s-child".into()
210 }
211 );
212
213 // Kind → Elternteil, SpawnedBy, beobachtet.
214 assert_eq!(child.edges.len(), 1);
215 assert_eq!(child.edges[0].kind, EdgeKind::SpawnedBy);
216 assert_eq!(child.edges[0].evidence, Evidence::Observed);
217 assert_eq!(
218 child.edges[0].to,
219 Endpoint::Session {
220 agent: "claude-code".into(),
221 local_id: "s-parent".into()
222 }
223 );
224}

255 r#"{"session_id":"c1","hook_event_name":"Stop"}"#,
256 2,

260 "codex",
261 None,
262 r#"{"session_id":"x1","hook_event_name":"Stop"}"#,
263 3,
264 );
265
266 let sessions = adapter::build(&journal).unwrap();
267 assert_eq!(sessions.len(), 2, "beide Agents sind erfasst");
268
269 // Sortiert nach (Agent, local_id): claude-code vor codex.
270 assert_eq!(sessions[0].agent.name, "claude-code");
271 assert_eq!(sessions[0].intent.request, "plane den Umbau");
272 assert_eq!(sessions[1].agent.name, "codex");
273 assert_eq!(sessions[1].intent.request, "review den Plan");
274
275 // codex hat (noch) keinen Normalisierer — der Prompt wird trotzdem erfasst,
276 // weil das Feld agent-unabhängig heißt; das Journal verliert nichts.
277 assert_eq!(sessions[1].lineage.as_ref().unwrap().local_id, "x1");
278}
