1//! Aus einem rohen [`JournalEvent`] die Fakten ziehen, die eine
2//! [`Session`](minds_core::Session) braucht — je Agent, was sich unterscheidet.
3//!
4//! Dieses Modul ist die Einlösung des Versprechens aus [`crate::hook_event`]:
5//! „heute ein gemeinsamer Umschlag für alle, morgen ein `match` je Agent". Der
6//! Umschlag (`session_id`, `transcript_path`, `cwd`, `hook_event_name`) ist bei
7//! allen Agents gleich und wird schon auf dem heißen Pfad normalisiert. Was
8//! *im* Payload steht — wie der Prompt-Text heißt, wie ein Tool-Aufruf seinen
9//! Pfad benennt — ist dagegen agent-spezifisch und wird erst hier, auf dem
10//! kalten Pfad, gedeutet.
11//!
12//! # Warum getrennt von `hook_event`
13//!
14//! [`hook_event::parse`](crate::hook_event::parse) läuft bei *jedem* Tool-Call
15//! im Prozess des Nutzers und darf deshalb nichts Teures tun. Es hält den
16//! Payload nur unverändert fest. Die Deutung — JSON-Felder herausparsen,
17//! Tool-Namen auf [`EffectKind`] abbilden — passiert später beim Checkpoint,
18//! wo Latenz niemandem wehtut. Derselbe Schnitt wie im ganzen Crate: heißer
19//! Pfad sammelt, kalter Pfad deutet.
20//!
21//! # Was der Hook liefert und was nicht
22//!
23//! Ein Hook-Payload trägt, was *im Moment des Ereignisses* bekannt ist: den
24//! Prompt bei `UserPromptSubmit`, Name und Eingabe eines Tools bei
25//! `PreToolUse`. Er trägt **nicht** den Antworttext des Modells — der steht nur
26//! im Transkript. Deshalb liefert dieses Modul bewusst nur die Hälfte: die
27//! andere Hälfte fügt der Adapter in M5.6 aus dem Transkript hinzu.
28//!
29//! # Robust gegen Unbekanntes
30//!
31//! Ein Payload, den wir nicht deuten können (fremder Agent, neues Tool,
32//! beschädigtes JSON), ergibt schlicht leere [`EventFacts`] — nie einen Fehler.
33//! Das Vokabular der Agents wächst schneller als unseres; ein unbekanntes Tool
34//! darf den Checkpoint einer ganzen Session nicht zum Absturz bringen. Was wir
35//! heute nicht deuten, liegt über den rohen Payload im Journal weiter bereit.
36

43/// Die gedeuteten Fakten eines einzelnen Hook-Events.
44///
45/// Alles ist `Option`: Ein `SessionStart` trägt weder Prompt noch Tool, ein
46/// unbekanntes Tool trägt kein [`Effect`]. Der Adapter fragt gezielt das ab,
47/// was der jeweilige [`EventKind`] erwarten lässt.
48#[derive(Debug, Clone, Default, PartialEq, Eq)]
49pub struct EventFacts {
50 /// Der Prompt-Text bei einem Prompt-Event.
51 pub prompt: Option<String>,
52
53 /// Der Tool-Aufruf bei einem Tool-Event.
54 pub tool: Option<ToolFacts>,
55}
56
57/// Ein normalisierter Tool-Aufruf, so weit der Hook ihn hergibt.
58#[derive(Debug, Clone, PartialEq, Eq)]
59pub struct ToolFacts {
60 /// Der Name des Tools in der Sprache des Agents (`Read`, `Bash`, …).
61 pub name: String,
62
63 /// Die rohe Tool-Eingabe als bereits serialisiertes JSON. Landet
64 /// unverändert in [`ToolCall::arguments`](minds_core::ToolCall::arguments),
65 /// damit der spätere Hash nicht von der Formatierung abhängt.
66 pub arguments: String,
67
68 /// Was der Aufruf in der Welt tat, normalisiert — Pfad und Art. Der
69 /// Inhalts-Hash bleibt hier `None`; ihn zu bilden ist M5.7.
70 pub effect: Option<Effect>,
71}
72
73/// Deutet ein Journal-Event für den benannten Agenten.
74///
75/// `agent` kommt aus dem [`SessionKey`](crate::SessionKey), nicht aus dem
76/// Payload — die Registrierung weiß besser, wer geschrieben hat, als das JSON.
77/// Ein unbekannter Agent ergibt leere Fakten; sein roher Payload bleibt im
78/// Journal erhalten.
79pub fn facts(agent: &str, event: &JournalEvent) -> EventFacts {
80 match event.kind {
81 // Der Prompt ist das eine wirklich agent-übergreifende Feld: Claude,
82 // Codex, Cursor und Gemini nennen ihn alle `prompt`. Ihn zu extrahieren
83 // braucht deshalb keinen agent-spezifischen Normalisierer — sonst
84 // verlöre ein noch nicht normalisierter Agent seine Prompts, und das
85 // Journal wäre umsonst „ein Beobachter für alle".
86 EventKind::Prompt => EventFacts {
87 prompt: parse::<Prompt>(event).and_then(|p| p.prompt),
88 tool: None,
89 },
90 // Die Tool→Effect-Abbildung ist dagegen agent-spezifisch (welcher
91 // Tool-Name schreibt, welches Feld trägt den Pfad) und dispatcht je
92 // Agent. Ein Agent ohne Normalisierer liefert hier `None`.
93 EventKind::ToolPre | EventKind::ToolPost => EventFacts {
94 prompt: None,
95 tool: tool_facts(agent, event),
96 },
97 _ => EventFacts::default(),
98 }
99}
100
101// ---------------------------------------------------------------------------
102// Tool-Normalisierung je Agent
103// ---------------------------------------------------------------------------
104
105/// Dispatch der Tool-Deutung. Weitere Agents kommen als eigene Zweige hinzu;
106/// jeder kennt nur die Felder *seines* Payloads.
107fn tool_facts(agent: &str, event: &JournalEvent) -> Option<ToolFacts> {
108 match agent {
109 "claude-code" => parse::<Tool>(event).and_then(claude_tool),
110 _ => None,
111 }
112}
113
114/// Baut aus Claude Codes `tool_name` + `tool_input` einen [`ToolFacts`].
115///
116/// Die Abbildung Tool-Name → [`EffectKind`] ist das agent-spezifische Wissen:
117/// Nur hier steht, dass `Edit` schreibt und `Bash` ausführt. Ein unbekanntes
118/// Tool ist [`EffectKind::Other`] ohne Pfad — kein Fehler, nur weniger Detail.
119fn claude_tool(t: Tool) -> Option<ToolFacts> {
120 let name = t.tool_name?;
121
122 // `tool_input` bleibt verbatim für `arguments`; den Effekt deutet die
123 // gemeinsame Abbildung.
124 let arguments = t
125 .tool_input
126 .as_ref()
127 .map(|r| r.get().to_owned())
128 .unwrap_or_default();
129 let effect = claude_effect(&name, t.tool_input.as_deref());
130
131 Some(ToolFacts {
132 name,
133 arguments,
134 effect: Some(effect),
135 })
136}
137
138/// Die agent-spezifische Abbildung Tool-Name + `tool_input` → [`Effect`] für
139/// Claude Code — geteilt vom Journal-Pfad ([`facts`]) und vom Transkript-Import
140/// ([`crate::import`]), damit „`Edit` schreibt, `Bash` führt aus" an genau einer
141/// Stelle steht.
142///
143/// `content` bleibt `None`; der Artefakt-Hash wird erst beim Checkpoint gebildet.
144/// Ein unbekanntes Tool ist [`EffectKind::Other`] ohne Pfad — kein Fehler, nur
145/// weniger Detail.
146pub fn claude_effect(tool_name: &str, tool_input: Option<&RawValue>) -> Effect {
147 let paths = tool_input
148 .and_then(|r| serde_json::from_str::<ToolPaths>(r.get()).ok())
149 .unwrap_or_default();
150
151 let (kind, path) = match tool_name {
152 "Read" => (EffectKind::Read, paths.file_path),
153 "Write" => (EffectKind::Write, paths.file_path),
154 "Edit" | "MultiEdit" => (EffectKind::Write, paths.file_path),
155 "NotebookEdit" => (EffectKind::Write, paths.notebook_path),
156 "Bash" => (EffectKind::Exec, None),
157 // Glob/Grep/WebFetch/Task/… greifen auf keinen einzelnen Pfad zu, den
158 // sich ein Artefakt-Hash merken könnte. Sie sind Teil der Erzählung,
159 // aber kein Datei-Effekt.
160 _ => (EffectKind::Other, None),
161 };
162
163 Effect {
164 kind,
165 path,
166 content: None,
167 }
168}

170/// Liest den Payload eines Events in `T`; misslingt das, ergibt es `None`
171/// statt eines Fehlers (siehe Modul-Doku).
172fn parse<T: for<'de> Deserialize<'de>>(event: &JournalEvent) -> Option<T> {
173 serde_json::from_str(event.payload.get()).ok()
174}

192/// tolerante Zweitdeutung des `tool_input`-Blocks. Alles Übrige wird ignoriert.
193#[derive(Debug, Default, Deserialize)]
194struct ToolPaths {
195 file_path: Option<String>,
196 notebook_path: Option<String>,
197}

Absicht

Ich möchte M5 so umbauen, dass wir es genauso machen wie entire! Mit dem Hook-Ansatz. Wir bleiben erst einmal bei der Rust implementierung. Diese ist effizienter. Das ist der entscheidende Unterschied. Minds soll auch Hooks installieren im Agenten selbst: Claude Code über `.claude/settings.json`, Codex über `.codex/hooks.json`, Cursor über `.cursor/hooks.json`, Gemini über `.gemini/settings.json`, OpenCode über ein TypeScript-Plugin. Bei Codex wird zusätzlich `codex_hooks = true` in die `config.toml` geschrieben. Dazu Git-Hooks: Checkpoints entstehen, wenn du oder der Agent committen; die Session-Metadaten möchte ich entweder auf einem child-repo haben oder falls wir diese kombination nicht haben direkt im repo. Das Verhalten soll an der Stelle absolut gleich sein.
