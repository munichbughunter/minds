1//! `minds enable` — Hooks im Agenten registrieren, idempotent und fremd-schonend.
2//!
3//! Das ist der Unterschied, der den Hook-Ansatz überhaupt trägt: Damit *jeder*
4//! Agent-Hook dasselbe `minds hook` aufruft, muss Minds sich in jedem Agenten
5//! eintragen. Fünf Agents, fünf Formate — die Fleißarbeit, die schon die Vision
6//! als schwerste Last benennt. Dieses Modul kapselt sie an *einer* Stelle.
7//!
8//! # Zwei eiserne Regeln
9//!
10//! 1. **Idempotent.** Ein zweites `minds enable` ändert nichts. Erkennbar an
11//! unserer Marke (`minds hook` in der Kommandozeile, ein Block zwischen
12//! [`MARK_BEGIN`]/[`MARK_END`] in Shell-Dateien).
13//! 2. **Fremdes bleibt.** Eine schon vorhandene Konfiguration in denselben
14//! Dateien — die eigenen Hooks des Nutzers, andere Tools — wird nie
15//! überschrieben. Wir fügen hinzu, wir ersetzen nicht.
16//!
17//! # Wohin was geschrieben wird
18//!
19//! | Agent | Datei |
20//! |-------------|--------------------------------------------------|
21//! | Claude Code | `.claude/settings.json` |
22//! | Codex | `.codex/hooks.json` + `codex_hooks` in `config.toml` |
23//! | Cursor | `.cursor/hooks.json` |
24//! | Gemini | `.gemini/settings.json` |
25//! | OpenCode | `.opencode/plugin/minds.ts` |
26//! | Git | `.git/hooks/post-commit`, `prepare-commit-msg` |
27//!
28//! Alles projekt-lokal, relativ zur Repo-Wurzel — der Kontext gehört zum Repo,
29//! nicht zum Benutzerkonto.
30//!
31//! # Ehrlich zu den Formaten
32//!
33//! Die JSON-Struktur der Claude-Code-Hooks ist belastbar. Für Codex, Cursor,
34//! Gemini und OpenCode folgt dieses Modul der dokumentierten Form zum Zeitpunkt
35//! des Umbaus (ADR-0003); ändert ein Agent sein Format, ist genau *diese* Stelle
36//! nachzuziehen — nicht der Rest des Systems. Deshalb steht das Format-Wissen je
37//! Agent gebündelt und benannt beisammen und nirgends sonst.

81 agent: Option<&str>,
82 store: &StoreConfig,
83 child_remote: Option<&str>,
84 verbose: bool,

87 Ok(paths) => paths,

100 "opencode" => &[Which::OpenCode],
101 "all" => Which::ALL,

142/// Registriert die gewählten Agents, immer die Git-Hooks (ein Checkpoint soll
143/// entstehen, wenn committet wird — unabhängig davon, welcher Agent lief) und
144/// schreibt die Store-Config.
145fn enable_agents(
146 paths: &RepoPaths,
147 agents: &[Which],
148 store: &StoreConfig,
149 child_remote: Option<&str>,
150 verbose: bool,
151) -> std::io::Result<()> {
152 for &agent in agents {
153 let change = match agent {
154 Which::ClaudeCode => claude_style(&paths.root, ".claude/settings.json", "claude-code")?,
155 Which::Codex => enable_codex(&paths.root)?,
156 Which::Cursor => claude_style(&paths.root, ".cursor/hooks.json", "cursor")?,
157 Which::Gemini => claude_style(&paths.root, ".gemini/settings.json", "gemini")?,
158 Which::OpenCode => enable_opencode(&paths.root)?,
159 };
160 report(verbose, agent_label(agent), change);
161 }
162
163 let post = enable_git_hook(&paths.git_dir, "post-commit", POST_COMMIT_BODY)?;
164 report(verbose, ".git/hooks/post-commit", post);
165 let prepare = enable_git_hook(&paths.git_dir, "prepare-commit-msg", PREPARE_MSG_BODY)?;
166 report(verbose, ".git/hooks/prepare-commit-msg", prepare);
167
168 // Der Kontext soll mit dem Code reisen — aber *woher* er reist, hängt am
169 // Backend. In-Repo liegt er im selben Repo; beim Child-Repo in einem
170 // separaten, das erst angelegt (oder geklont) werden muss.
171 configure_sync(paths, store, child_remote, verbose)?;
172
173 // Die Store-Config gehört zum Setup: Ohne sie wüssten checkpoint/show/why
174 // nicht, wo der Kontext liegt. `git config` setzt idempotent.
175 config::write(&paths.root, store)?;
176 vln(verbose, " .git/config: minds.backend/contextRef gesetzt");
177
178 // Der Backfill läuft losgelöst im Hintergrund: Wer Minds spät einrichtet,
179 // soll rückwirkend Kontext bekommen, ohne auf das Lesen (womöglich großer)
180 // Transkripte zu warten. Ausgabe und Fehler landen im Log, nicht im
181 // Terminal des Nutzers.
182 spawn_background_import(&paths.git_dir);
183 vln(
184 verbose,
185 " Backfill läuft im Hintergrund → .git/minds/import.log",
186 );
187
188 Ok(())
189}

198/// Startet den Backfill als **losgelösten** Prozess: dasselbe Binary, mit dem
199/// versteckten Flag, stdout/stderr in eine Log-Datei, kein `wait`.
200///
201/// Best effort: Lässt sich der Prozess nicht starten (kein `current_exe`, kein
202/// Log), bleibt das Setup trotzdem erfolgreich — der Backfill ist eine
203/// Zugabe, kein Kernschritt. Er kann jederzeit durch ein erneutes `minds
204/// enable` angestoßen werden.
205fn spawn_background_import(git_dir: &Path) {
206 let Ok(exe) = std::env::current_exe() else {
207 return;
208 };
209
210 let log_dir = git_dir.join("minds");
211 let _ = fs::create_dir_all(&log_dir);
212 let log = fs::OpenOptions::new()
213 .create(true)
214 .append(true)
215 .open(log_dir.join("import.log"));
216
217 let mut cmd = Command::new(exe);
218 cmd.arg("enable")
219 .arg(BACKGROUND_IMPORT_FLAG)
220 .stdin(Stdio::null());
221
222 match log {
223 Ok(file) => {
224 let err = file.try_clone().ok();
225 cmd.stdout(Stdio::from(file));
226 if let Some(err) = err {
227 cmd.stderr(Stdio::from(err));
228 }
229 }
230 // Ohne Log lieber still als lärmend im Terminal.
231 Err(_) => {
232 cmd.stdout(Stdio::null()).stderr(Stdio::null());
233 }
234 }
235
236 // Absichtlich kein `wait`: Der Prozess überlebt `minds enable` und wird von
237 // init adoptiert.
238 let _ = cmd.spawn();
239}

241/// Fügt einen **additiven** Fetch-Refspec für den Kontext-Ref an `origin` an,
242/// sodass `git fetch` ihn mitzieht. Anders als beim Push ist das gefahrlos:
243/// Fetch-Refspecs sind eine Menge, kein Ersatz.
244///
245/// Nur wenn `origin` existiert; ein Repo ohne Remote bekommt den Refspec beim
246/// nächsten `minds enable` nach `git remote add`. Idempotent: ist der Refspec
247/// schon da, ändert sich nichts.
248fn ensure_fetch_refspec(root: &Path) -> std::io::Result<bool> {
249 if !git_output(root, &["remote"])?
250 .lines()
251 .any(|r| r.trim() == "origin")
252 {
253 return Ok(false);
254 }
255
256 let spec = "+refs/minds/context:refs/minds/context";
257 let existing = git_output(root, &["config", "--get-all", "remote.origin.fetch"])?;
258 if existing.lines().any(|line| line.trim() == spec) {
259 return Ok(false);
260 }
261
262 let status = Command::new("git")
263 .arg("-C")
264 .arg(root)
265 .args(["config", "--add", "remote.origin.fetch", spec])
266 .status()?;
267 Ok(status.success())
268}

285/// Richtet den Push/Fetch des Kontext-Refs ein — für In-Repo im selben
286/// Repository, für das Child-Repo im separaten, das dafür erst existieren muss.
287fn configure_sync(
288 paths: &RepoPaths,
289 store: &StoreConfig,
290 child_remote: Option<&str>,
291 verbose: bool,
292) -> std::io::Result<()> {
293 match store.backend() {
294 Backend::InRepo => {
295 let pre_push = enable_git_hook(&paths.git_dir, "pre-push", PRE_PUSH_BODY_IN_REPO)?;
296 report(verbose, ".git/hooks/pre-push", pre_push);
297 if ensure_fetch_refspec(&paths.root)? {
298 vln(
299 verbose,
300 " .git/config: Fetch-Refspec für refs/minds/context gesetzt",
301 );
302 }
303 }
304 Backend::ChildRepo { path } => {
305 let child = resolve_against(&paths.root, path);
306 match ensure_child_repo(&paths.root, &child, child_remote) {
307 Ok(change) => report(verbose, &format!("Child-Repo {}", child.display()), change),
308 // Ein Fehler beim Child-Repo kommt immer durch — ohne das Repo
309 // kann checkpoint nicht speichern.
310 Err(err) => eprintln!(" Child-Repo {}: {err}", child.display()),
311 }
312 let pre_push =
313 enable_git_hook(&paths.git_dir, "pre-push", &pre_push_child_body(&child))?;
314 report(verbose, ".git/hooks/pre-push (child)", pre_push);
315 }
316 }
317 Ok(())
318}

331/// Legt das Child-Repo an oder klont es — oder lässt es, wenn es schon existiert.
332///
333/// - Mit `--child-remote`: `git clone --bare <url>`; scheitert das (unerreichbar,
334/// noch leer angelegt), wird ein bares Repo initialisiert und `origin` gesetzt,
335/// damit der spätere Push trotzdem ein Ziel hat.
336/// - Ohne Remote: ein bares Repo, rein lokal.
337///
338/// In beiden Fällen bekommt das Child-Repo eine Committer-Identität (vom
339/// Code-Repo geerbt, sonst ein Default) — ohne die kann der Store dort keinen
340/// Commit schreiben.
341fn ensure_child_repo(root: &Path, child: &Path, remote: Option<&str>) -> std::io::Result<Change> {
342 if is_git_repo(child) {
343 if let Some(url) = remote {
344 ensure_child_origin(child, url)?;
345 }
346 return Ok(Change::Unchanged);
347 }
348
349 if let Some(parent) = child.parent() {
350 fs::create_dir_all(parent)?;
351 }
352
353 match remote {
354 Some(url) if git_ok(&["clone", "--bare", "--quiet", url, &child.to_string_lossy()]) => {}
355 Some(url) => {
356 git_init_bare(child)?;
357 let _ = run_git(child, &["remote", "add", "origin", url]);
358 }
359 None => git_init_bare(child)?,
360 }
361
362 set_child_identity(root, child);
363 Ok(Change::Created)
364}

390/// Sorgt dafür, dass `origin` des Child-Repos auf `url` zeigt — hinzufügen oder
391/// umbiegen, idempotent.
392fn ensure_child_origin(child: &Path, url: &str) -> std::io::Result<()> {
393 let current = git_output(child, &["remote", "get-url", "origin"]).unwrap_or_default();
394 let current = current.trim();
395 if current == url {
396 return Ok(());
397 }
398 if current.is_empty() {
399 let _ = run_git(child, &["remote", "add", "origin", url]);
400 } else {
401 let _ = run_git(child, &["remote", "set-url", "origin", url]);
402 }
403 Ok(())
404}

475/// Trägt in eine Claude-artige `hooks`-Map je Event einen Command-Hook ein, der
476/// `minds hook --agent <agent>` aufruft — idempotent und ohne Fremdes zu
477/// verwerfen.
478fn claude_style(root: &Path, rel: &str, agent: &str) -> std::io::Result<Change> {
479 let path = root.join(rel);
480 let mut root_value = read_json(&path)?;
481 let existed = path.exists();
482
483 let obj = as_object(&mut root_value);
484 let hooks = obj
485 .entry("hooks")
486 .or_insert_with(|| Value::Object(Map::new()));
487 let hooks = match hooks.as_object_mut() {
488 Some(map) => map,
489 // Eine fremde, nicht-Objekt-`hooks` würden wir sonst zertrümmern.
490 None => return Ok(Change::Unchanged),
491 };
492
493 let mut changed = false;
494 for &(event, matcher) in HOOK_EVENTS {
495 let command = command_for(agent, event, agent != "claude-code");
496 let groups = hooks
497 .entry(event)
498 .or_insert_with(|| Value::Array(Vec::new()));
499 let Some(groups) = groups.as_array_mut() else {
500 continue;
501 };
502 if array_has_minds_hook(groups) {
503 continue;
504 }
505 groups.push(hook_group(matcher, &command));
506 changed = true;
507 }
508
509 if !changed && existed {
510 return Ok(Change::Unchanged);
511 }
512 write_json(&path, &root_value)?;
513 Ok(if existed {
514 Change::Updated
515 } else {
516 Change::Created
517 })
518}

520/// `{"type":"command","command":"minds hook …"}`, ggf. in eine Matcher-Gruppe
521/// verpackt.
522fn hook_group(matcher: bool, command: &str) -> Value {
523 let entry = json!({ "type": "command", "command": command });
524 if matcher {
525 json!({ "matcher": "*", "hooks": [entry] })
526 } else {
527 json!({ "hooks": [entry] })
528 }
529}

531/// `true`, wenn irgendeine Gruppe schon einen `minds hook`-Command trägt — die
532/// Idempotenz-Prüfung. Bewusst über die Teilzeichenkette, damit auch eine
533/// ältere, leicht anders geschriebene Registrierung als „schon da" gilt.
534fn array_has_minds_hook(groups: &[Value]) -> bool {
535 groups.iter().any(|group| {
536 group
537 .get("hooks")
538 .and_then(Value::as_array)
539 .into_iter()
540 .flatten()
541 .filter_map(|h| h.get("command").and_then(Value::as_str))
542 .any(|cmd| cmd.contains("minds hook"))
543 })
544}

560/// Codex braucht zweierlei: die Hook-Registrierung *und* den Schalter
561/// `codex_hooks = true`, ohne den Codex die `hooks.json` gar nicht liest.
562fn enable_codex(root: &Path) -> std::io::Result<Change> {
563 let hooks_change = claude_style(root, ".codex/hooks.json", "codex")?;
564 let toml_change = ensure_codex_hooks_flag(&root.join(".codex/config.toml"))?;
565 // Der interessantere der beiden Änderungsstände bestimmt den Bericht.
566 Ok(match (hooks_change, toml_change) {
567 (Change::Created, _) | (_, Change::Created) => Change::Created,
568 (Change::Updated, _) | (_, Change::Updated) => Change::Updated,
569 _ => Change::Unchanged,
570 })
571}

573/// Setzt `codex_hooks = true` in `config.toml` — zeilenbasiert, ohne
574/// TOML-Abhängigkeit.
575///
576/// Bewusst schlicht: Steht die Zeile schon (auf `true`), passiert nichts. Steht
577/// sie auf `false` oder fehlt sie, wird sie gesetzt bzw. angehängt. Das deckt
578/// den Alltag; eine `codex_hooks` tief in einer `[table]` verschachtelt käme in
579/// echtem TOML vor, ist hier aber nicht der Fall (der Schalter ist top-level).
580fn ensure_codex_hooks_flag(path: &Path) -> std::io::Result<Change> {
581 let existed = path.exists();
582 let current = if existed {
583 fs::read_to_string(path)?
584 } else {
585 String::new()
586 };
587
588 let mut lines: Vec<String> = current.lines().map(str::to_owned).collect();
589 let mut found = false;
590 let mut changed = false;
591 for line in &mut lines {
592 if line.trim_start().starts_with("codex_hooks") {
593 found = true;
594 if line.trim() != "codex_hooks = true" {
595 *line = "codex_hooks = true".to_string();
596 changed = true;
597 }
598 break;
599 }
600 }
601 if !found {
602 lines.push("codex_hooks = true".to_string());
603 changed = true;
604 }
605
606 if !changed && existed {
607 return Ok(Change::Unchanged);
608 }
609 create_parent(path)?;
610 let mut out = lines.join("\n");
611 out.push('\n');
612 fs::write(path, out)?;
613 Ok(if existed {
614 Change::Updated
615 } else {
616 Change::Created
617 })
618}

680/// pre-push für das **In-Repo**-Backend: schickt den Kontext-Ref beim `git push`
681/// **zusätzlich** mit, an dasselbe Remote, an das gerade gepusht wird (`$1`).

683/// Warum ein Hook und kein `remote.push`-Refspec: Sobald `remote.<name>.push`
684/// gesetzt ist, pusht `git push` **nur noch** diese Refspecs — der Branch des
685/// Nutzers bliebe liegen. Der Hook lässt den normalen Push unangetastet und
686/// legt den Kontext daneben.

688/// `MINDS_SYNCING` bricht die Rekursion: Der innere Push löst pre-push erneut
689/// aus, tut dann aber nichts. Gepusht wird nur, wenn es den Ref überhaupt gibt;

691/// Nutzers nie verhindert.

693git rev-parse --verify --quiet refs/minds/context >/dev/null 2>&1; then \
