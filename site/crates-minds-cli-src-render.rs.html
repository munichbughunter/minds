11/// Führt die Verweise eines Commits zusammen: die Trailer (verbindlich,
12/// [`Evidence::Observed`]) und die Kanten aus dem Store-Index (heuristisch). Der
13/// Trailer gewinnt — eine Session, die schon beobachtet verknüpft ist, wird
14/// nicht zusätzlich als vermutet geführt.
15///
16/// Rein und ohne I/O, damit die Zusammenführung ohne Repository prüfbar ist.
17pub fn merge_links(
18 trailers: &[SessionId],
19 index_links: &[IndexLink],
20) -> Vec<(SessionId, Evidence)> {
21 let mut out: Vec<(SessionId, Evidence)> = trailers
22 .iter()
23 .map(|id| (*id, Evidence::Observed))
24 .collect();
25 for link in index_links {
26 if !out.iter().any(|(id, _)| *id == link.session) {
27 out.push((link.session, link.evidence));
28 }
29 }
30 out
31}

43/// Zeichnet die Sessions eines Commits als Baum unter `header` — angelehnt an
44/// Werkzeuge wie Dagger: eine Überschrift, darunter je Session ein Ast mit dem
45/// Intent als Blickfang und der Herkunft dezent darunter.
46///
47/// **Kompakt** im Regelfall (eine Intent-Zeile, eine Metazeile, die Zahl der
48/// Dateien). `full` klappt alles auf: ganzer Prompt, alle Dateien, Constraints,
49/// verworfene Pfade, Kanten.
50pub fn tree(header: &str, items: &[Shown], full: bool) {
51 println!("{}", bold(header));
52
53 let n = items.len();
54 for (i, item) in items.iter().enumerate() {
55 let last = i + 1 == n;
56 let branch = if last { "╰─" } else { "├─" };
57 let cont = if last { " " } else { "│ " };
58 let s = item.session;
59
60 // Der Intent ist der Blickfang.
61 let request = if full {
62 s.intent.request.clone()
63 } else {
64 headline(&s.intent.request, 96)
65 };
66 println!("{branch} {}", bold(&format!("▸ {request}")));
67
68 // Herkunft, dezent.
69 let files = s.produced.files.len();
70 let fileword = if files == 1 { "Datei" } else { "Dateien" };
71 println!(
72 "{cont}{}",
73 dim(&format!(
74 "{} {} · {}/{} · {}/{} Token · {files} {fileword}",
75 s.agent.name,
76 s.agent.version,
77 s.model.provider,
78 s.model.id,
79 s.usage.input_tokens,
80 s.usage.output_tokens,
81 ))
82 );
83 let vermerk = if item.evidence == Evidence::Observed {
84 String::new()
85 } else {
86 format!(" ({})", evidence(item.evidence))
87 };
88 println!(
89 "{cont}{}",
90 dim(&format!("{}{vermerk}", short_id(item.id, full)))
91 );
92
93 if full {
94 for constraint in &s.intent.constraints {
95 println!("{cont} Constraint: {constraint}");
96 }
97 for discarded in &s.intent.discarded {
98 println!("{cont} Verworfen: {discarded}");
99 }
100 for file in &s.produced.files {
101 println!("{cont} {file}");
102 }
103 for edge in &s.edges {
104 println!(
105 "{cont} Kante: {} → {} ({})",
106 edge_kind(edge.kind),
107 endpoint(&edge.to),
108 evidence(edge.evidence),
109 );
110 }
111 } else if files > 0 {
112 println!(
113 "{cont}{}",
114 dim("minds show --full zeigt Dateien und Prompt")
115 );
116 }
117 }
118}

120/// Löst die Verweise eines Commits gegen den Store auf und zeichnet den Baum —
121/// der gemeinsame Weg von [`crate::show`] und [`crate::why`].
122///
123/// Übersprungen wird, was keinen Mehrwert hat: Sessions **ohne erfasste
124/// Absicht** (der frühere „(kein Prompt erfasst)"-Ballast). Ein Verweis auf eine
125/// Session, die der Store nicht hat, wird als Waise gezählt und dezent
126/// vermerkt — `minds fsck` geht dem systematisch nach.
127pub fn show_links(
128 header: &str,
129 links: &[(SessionId, Evidence)],
130 store: &dyn minds_store::ContextStore,
131 full: bool,
132) -> Result<(), Box<dyn std::error::Error>> {
133 let mut owned: Vec<(SessionId, Session, Evidence)> = Vec::new();
134 let mut orphans = 0usize;
135 for (id, evidence) in links {
136 match store.get(*id)? {
137 Some(session) if !session.intent.request.trim().is_empty() => {
138 owned.push((*id, session, *evidence))
139 }
140 // Ohne Absicht kein Eintrag — stumm weglassen.
141 Some(_) => {}
142 None => orphans += 1,
143 }
144 }
145
146 if owned.is_empty() {
147 println!("{}", bold(header));
148 let note = if orphans > 0 {
149 format!("{orphans} Verweis(e) ins Leere — siehe minds fsck")
150 } else {
151 "kein Minds-Kontext für diesen Commit".to_string()
152 };
153 println!(" {}", dim(&note));
154 return Ok(());
155 }
156
157 let shown: Vec<Shown> = owned
158 .iter()
159 .map(|(id, session, evidence)| Shown {
160 id: *id,
161 session,
162 evidence: *evidence,
163 })
164 .collect();
165 tree(header, &shown, full);
166
167 if orphans > 0 {
168 println!(" {}", dim(&format!("+ {orphans} Verweis(e) ins Leere")));
169 }
170 Ok(())
171}
