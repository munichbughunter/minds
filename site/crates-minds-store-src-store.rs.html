1//! Der [`ContextStore`]-Trait: put, get, exists, list — nach [`SessionId`].
2//!
3//! # Die ID kommt nie von außen
4//!
5//! [`ContextStore::put`] nimmt keine ID entgegen, es *gibt* eine zurück. Der
6//! Schlüssel ist der Hash des Inhalts (Architektur-Prinzip 1), und ein
7//! Aufrufer, der ihn mitliefern dürfte, könnte lügen — mit dem Ergebnis, dass
8//! `minds why` einen Trailer auflöst und eine fremde Session zeigt. Ein
9//! Audit-Record, dessen Schlüssel und Inhalt auseinanderlaufen können, ist
10//! keiner.
11//!
12//! Auf dem Lesepfad prüft [`ContextStore::get`] deshalb nach: Was unter der ID
13//! liegt, muss auf sie hashen. Das kostet einen blake3-Durchlauf über wenige
14//! Kilobyte und macht aus der Content-Adressierung einen Selbsttest — wer eine
15//! Session im Store nachträglich editiert, fliegt beim nächsten Lesen auf.
16//!
17//! # Dedup ist Abwesenheit von Arbeit
18//!
19//! Gleicher Inhalt ⇒ gleiche ID ⇒ derselbe Ort. Ein Backend muss dafür nichts
20//! tun, es darf es nur nicht kaputtmachen; [`Put::AlreadyPresent`] ist der
21//! Beleg, dass ein wiederholtes `put` nichts geschrieben hat. Das ist genau die
22//! Zusage, die der nächste Commit (`idempotentes put`) für den `InRepoStore`
23//! einlöst.
24//!
25//! # Was `get` nicht prüft
26//!
27//! Es wird **nicht** neu kanonisiert und mit dem Gespeicherten verglichen. Das
28//! wäre die schärfere Prüfung — und die falsche: Eine Session aus einer
29//! neueren Schema-Version trägt Felder, die dieses Binary nicht kennt und beim
30//! Deserialisieren verwirft (Vorwärts-Toleranz, Architektur-Prinzip 4). Neu
31//! serialisiert käme etwas Kürzeres heraus, und ein alter Reader würde jede
32//! neuere Session für beschädigt erklären. Geprüft wird deshalb gegen die
33//! **gespeicherten Bytes**, nicht gegen das, was wir daraus wieder machen
34//! würden.

43/// Was [`ContextStore::put`] bewirkt hat.
44///
45/// Für den Aufrufer selten handlungsrelevant, für die CLI-Ausgabe und das
46/// Protokoll aber der Unterschied zwischen „erfasst" und „kannten wir schon".
47/// Gleiche Bauform wie `minds_git::RefUpdate` eine Ebene tiefer.
48#[derive(Debug, Clone, Copy, PartialEq, Eq)]
49pub enum Put {
50 /// Die Session lag noch nicht im Store und wurde geschrieben.
51 Written(SessionId),
52 /// Die Session lag bereits unter dieser ID — nichts geschrieben.
53 AlreadyPresent(SessionId),
54}

60 Put::Written(id) | Put::AlreadyPresent(id) => *id,

65 pub fn was_written(&self) -> bool {
66 matches!(self, Put::Written(_))

70/// Ein Speicher für Sessions, adressiert über ihre [`SessionId`].
71///
72/// Zu implementieren sind nur die drei Byte-Methoden; [`put`](Self::put),
73/// [`get`](Self::get) und [`exists`](Self::exists) kommen vom Trait und gelten
74/// damit für jedes Backend gleich (siehe Crate-Doku).
75///
76/// Der Trait ist objekt-sicher: Die CLI wählt ihr Backend zur Laufzeit aus der
77/// Konfiguration und hält es als `Box<dyn ContextStore>`.
78pub trait ContextStore {
79 /// Legt eine Session ab, wenn sie noch nicht da ist.
80 ///
81 /// Muss idempotent sein: Liegt unter `session.id()` bereits etwas, wird
82 /// nichts geschrieben und [`Put::AlreadyPresent`] zurückgegeben. Weil die ID
83 /// der Hash des Inhalts ist, kann „liegt bereits da" nichts anderes heißen
84 /// als „liegt bereits *so* da".
85 fn put_bytes(&self, session: &SessionBytes) -> Result<Put>;
86
87 /// Holt die rohen, gespeicherten Bytes — `None`, wenn nichts unter `id`
88 /// liegt.
89 ///
90 /// Ungeprüft: weder gehasht noch geparst. Das ist der Durchreich-Pfad für
91 /// den Reader, der das JSON unverändert weitergibt. Wer eine
92 /// [`Session`] will, nimmt [`get`](Self::get).
93 fn get_bytes(&self, id: SessionId) -> Result<Option<Vec<u8>>>;
94
95 /// Alle IDs im Store — sortiert und ohne Dopplungen.
96 ///
97 /// Die Sortierung ist eine Zusage an Tests und an den Reader-Index: Eine
98 /// Reihenfolge, die von der Traversierung des Backends abhängt, wäre für
99 /// beide unbrauchbar.
100 fn list(&self) -> Result<Vec<SessionId>>;
101
102 /// Die rohen Bytes des Commit-Index (`index.json`) — `None`, wenn keiner
103 /// abgelegt ist.
104 ///
105 /// Der Index ist der einzige Nachbar der Sessions im Baum (siehe
106 /// [`crate::index`] und [`crate::layout`]). Er wird über denselben Ref
107 /// geschrieben und reist deshalb beim Push mit.
108 fn get_index_bytes(&self) -> Result<Option<Vec<u8>>>;
109
110 /// Schreibt die rohen Bytes des Commit-Index.
111 fn put_index_bytes(&self, bytes: &[u8]) -> Result<()>;
112
113 /// Ob unter `id` etwas liegt.
114 ///
115 /// Der Default liest den Inhalt und wirft ihn weg — korrekt, aber
116 /// verschwenderisch. Ein Backend, das billiger nachsehen kann (ein
117 /// Baum-Lookup ohne den Blob zu lesen), überschreibt das.
118 fn exists(&self, id: SessionId) -> Result<bool> {
119 Ok(self.get_bytes(id)?.is_some())
120 }
121
122 /// Der Commit-Index, geparst.
123 ///
124 /// Fehlt er, ist er leer. Ist er beschädigt, ebenfalls leer statt ein Fehler:
125 /// Ein kaputter Index darf `minds show` nicht abschießen — er ist eine
126 /// heuristische Ergänzung, kein tragendes Teil. `minds fsck` ist der Ort, dem
127 /// nachzugehen.
128 fn index(&self) -> Result<CommitIndex> {
129 match self.get_index_bytes()? {
130 Some(bytes) => Ok(serde_json::from_slice(&bytes).unwrap_or_default()),
131 None => Ok(CommitIndex::default()),
132 }
133 }
134
135 /// Legt den Commit-Index ab.
136 fn set_index(&self, index: &CommitIndex) -> Result<()> {
137 // Ein Index aus Strings und Enums serialisiert immer; ein Fehler hier
138 // wäre ein Bug im Typ, kein Laufzeitzustand.
139 let bytes = serde_json::to_vec(index).expect("CommitIndex serialisiert immer");
140 self.put_index_bytes(&bytes)
141 }
142
143 /// Legt eine geprüfte Session ab und gibt ihre ID zurück.
144 ///
145 /// Der reguläre Schreibweg. Kanonisiert, hasht und reicht an
146 /// [`put_bytes`](Self::put_bytes) weiter — mehr passiert nicht, und genau
147 /// deshalb ist es an einer Stelle richtig statt in jedem Backend erneut.
148 ///
149 /// Dass hier eine [`RedactedSession`] verlangt wird und keine [`Session`],
150 /// ist die fail-closed-Zusage in Typform (siehe Crate-Doku).
151 fn put(&self, session: &RedactedSession) -> Result<Put> {
152 self.put_bytes(&SessionBytes::of(session)?)
153 }
154
155 /// Holt die Session unter `id` — `None`, wenn sie hier nicht liegt.
156 ///
157 /// `None` heißt „nicht in diesem Store", nicht „gibt es nicht": Beim
158 /// Child-Repo-Backend kann derselbe Trailer auflösbar werden, sobald der
159 /// Kontext-Ref gefetcht ist. Das ist die *graceful degradation* aus dem Plan
160 /// — ein leeres Ergebnis, kein harter Fehler.
161 ///
162 /// # Fehler
163 ///
164 /// - [`StoreError::Corrupt`] — der Inhalt hasht nicht auf `id`.
165 /// - [`StoreError::Malformed`] — der Inhalt ist kein gültiges Session-JSON.
166 /// - [`StoreError::Unredacted`] — die Session ist nicht als redigiert
167 /// markiert. Sie stammt dann nicht aus diesem Werkzeug; sie an den Reader
168 /// zu geben hieße, ungeprüften Text in eine HTML-Seite zu rendern. Wer die
169 /// Bytes trotzdem braucht (Forensik), nimmt
170 /// [`get_bytes`](Self::get_bytes).
171 fn get(&self, id: SessionId) -> Result<Option<Session>> {
172 let Some(bytes) = self.get_bytes(id)? else {
173 return Ok(None);
174 };
175
176 let actual = SessionId::from_canonical_bytes(&bytes);
177 if actual != id {
178 return Err(StoreError::Corrupt {
179 requested: id,
180 actual,
181 });
182 }
183
184 let session: Session =
185 serde_json::from_slice(&bytes).map_err(|err| StoreError::malformed(id, err))?;
186
187 if !session.redaction.applied {
188 return Err(StoreError::Unredacted { id });
189 }
190
191 Ok(Some(session))
192 }
193}

205 /// Ein Store im Speicher — Referenz-Implementierung für den Trait-Vertrag.
206 ///
207 /// Er hält genau das, was jedes Backend hält: `SessionId → Bytes`. Was diese
208 /// Tests prüfen, sind deshalb die Zusagen des *Traits*, nicht die von Git;

216 impl MemoryStore {
217 /// Schreibt am Trait vorbei — nur so lässt sich ein beschädigter Store
218 /// herstellen, den es sonst nicht geben kann.
219 fn insert_raw(&self, id: SessionId, bytes: impl Into<Vec<u8>>) {
220 self.entries.borrow_mut().insert(id, bytes.into());
221 }

224 impl ContextStore for MemoryStore {
225 fn put_bytes(&self, session: &SessionBytes) -> Result<Put> {
226 let mut entries = self.entries.borrow_mut();
227 if entries.contains_key(&session.id()) {
228 return Ok(Put::AlreadyPresent(session.id()));
229 }
230 entries.insert(session.id(), session.as_bytes().to_vec());
231 Ok(Put::Written(session.id()))
232 }
233
234 fn get_bytes(&self, id: SessionId) -> Result<Option<Vec<u8>>> {
235 Ok(self.entries.borrow().get(&id).cloned())
236 }
237
238 fn list(&self) -> Result<Vec<SessionId>> {
239 // BTreeMap iteriert sortiert — die Zusage des Traits fällt hier
240 // gratis ab.
241 Ok(self.entries.borrow().keys().copied().collect())
242 }
243
244 fn get_index_bytes(&self) -> Result<Option<Vec<u8>>> {
245 Ok(self.index.borrow().clone())
246 }
247
248 fn put_index_bytes(&self, bytes: &[u8]) -> Result<()> {
249 *self.index.borrow_mut() = Some(bytes.to_vec());
250 Ok(())
251 }
252 }

Absicht

Ich möchte M5 so umbauen, dass wir es genauso machen wie entire! Mit dem Hook-Ansatz. Wir bleiben erst einmal bei der Rust implementierung. Diese ist effizienter. Das ist der entscheidende Unterschied. Minds soll auch Hooks installieren im Agenten selbst: Claude Code über `.claude/settings.json`, Codex über `.codex/hooks.json`, Cursor über `.cursor/hooks.json`, Gemini über `.gemini/settings.json`, OpenCode über ein TypeScript-Plugin. Bei Codex wird zusätzlich `codex_hooks = true` in die `config.toml` geschrieben. Dazu Git-Hooks: Checkpoints entstehen, wenn du oder der Agent committen; die Session-Metadaten möchte ich entweder auf einem child-repo haben oder falls wir diese kombination nicht haben direkt im repo. Das Verhalten soll an der Stelle absolut gleich sein.
