1//! Das Rendern — reine Funktionen von Daten nach HTML-Text.
2//!
3//! Kein Template-Modul, keine Abhängigkeit: Die Seite ist klein genug, um sie
4//! mit `format!` zu schreiben, und das hält den Reader bei genau den drei
5//! Crates, die er ohnehin braucht.
6//!
7//! # Selbsttragend
8//!
9//! Jede Seite bringt ihr CSS und ihr JavaScript **inline** mit. Kein CDN, keine
10//! Schriftart aus dem Netz, kein Build-Schritt. Damit funktioniert die Ausgabe
11//! auch über `file://`, im Air-Gap und hinter jeder Firewall — dieselbe
12//! Bedingung, unter der die Kundschaft aus der Vision arbeitet.
13//!
14//! # Warum die Panels in Rust vorgerendert werden
15//!
16//! Der naheliegende Weg wäre, die Sessions als JSON einzubetten und im Browser
17//! zu einem Panel zu bauen. Das hieße: ein zweites Datenformat, ein
18//! Template-Dialekt in JavaScript und eine Abhängigkeit auf `serde_json`.
19//! Stattdessen wird jedes Session-Panel **einmal in Rust** gerendert und fertig
20//! in die Seite gelegt; das Skript tut nur noch zweierlei — beim Laden alle
21//! verstecken und auf Klick das richtige zeigen.
22//!
23//! Die Reihenfolge ist Absicht: Ausgeliefert wird **sichtbar**, versteckt wird
24//! erst im Browser. Ohne JavaScript steht damit jede Session lesbar unter dem
25//! Code, statt dass die Seite leer wirkt — progressive Verbesserung, keine
26//! Voraussetzung.

90/// Der Codeblock: je Zeile eine Zeilennummer und der Text. Zeilen mit Kontext
91/// tragen `data-sessions` und werden dadurch klickbar.
92fn code_block(view: &FileView) -> String {
93 let mut out = String::new();
94 for line in &view.lines {
95 let ids = line
96 .sessions
97 .iter()
98 .map(|id| id.to_string())
99 .collect::<Vec<_>>()
100 .join(" ");
101
102 if line.is_attributed() {
103 let count = line.sessions.len();
104 let title = if count == 1 {
105 "Session hinter dieser Zeile anzeigen".to_string()
106 } else {
107 format!("{count} Sessions hinter dieser Zeile anzeigen")
108 };
109 out.push_str(&format!(
110 "<div class=\"line has-context\" data-sessions=\"{ids}\" tabindex=\"0\" \
111 role=\"button\" title=\"{title}\">\
112 <span class=\"num\">{num}</span><code>{text}</code></div>\n",
113 ids = escape(&ids),
114 title = escape(&title),
115 num = line.number,
116 text = escape(&line.text),
117 ));
118 } else {
119 out.push_str(&format!(
120 "<div class=\"line\"><span class=\"num\">{num}</span><code>{text}</code></div>\n",
121 num = line.number,
122 text = escape(&line.text),
123 ));
124 }
125 }
126 out
127}

154pub fn session_panel(id: SessionId, session: &Session, inferred: bool) -> String {
155 let badge = if inferred {
156 "<p class=\"guess\">⚠ vermutet — heuristisch verknüpft, nicht über einen Trailer belegt</p>\n"
157 } else {
158 ""
159 };
160 let mut out = format!(
161 "<section class=\"session\" id=\"s-{id}\">\n\
162 {badge}\
163 <h2>Absicht</h2>\n\
164 <p class=\"intent\">{request}</p>\n",
165 id = escape(&id.to_string()),
166 request = escape(non_empty(&session.intent.request, "(kein Prompt erfasst)")),
167 );
168
169 if !session.intent.constraints.is_empty() {
170 out.push_str("<h3>Constraints</h3>\n<ul>\n");
171 for constraint in &session.intent.constraints {
172 out.push_str(&format!("<li>{}</li>\n", escape(constraint)));
173 }
174 out.push_str("</ul>\n");
175 }
176
177 if !session.intent.discarded.is_empty() {
178 out.push_str("<h3>Verworfen</h3>\n<ul>\n");
179 for discarded in &session.intent.discarded {
180 out.push_str(&format!("<li>{}</li>\n", escape(discarded)));
181 }
182 out.push_str("</ul>\n");
183 }
184
185 out.push_str(&format!(
186 "<h3>Herkunft</h3>\n\
187 <dl>\n\
188 <dt>Agent</dt><dd>{agent} {version}</dd>\n\
189 <dt>Modell</dt><dd>{provider} / {model}</dd>\n\
190 <dt>Tokens</dt><dd>{input} ein / {output} aus</dd>\n\
191 <dt>Session</dt><dd><code class=\"id\">{id}</code></dd>\n\
192 </dl>\n",
193 agent = escape(&session.agent.name),
194 version = escape(&session.agent.version),
195 provider = escape(&session.model.provider),
196 model = escape(&session.model.id),
197 input = session.usage.input_tokens,
198 output = session.usage.output_tokens,
199 id = escape(&id.to_string()),
200 ));
201
202 if !session.produced.files.is_empty() {
203 out.push_str("<h3>Berührte Dateien</h3>\n<ul>\n");
204 for file in &session.produced.files {
205 out.push_str(&format!("<li><code>{}</code></li>\n", escape(file)));
206 }
207 out.push_str("</ul>\n");
208 }
209
210 out.push_str("</section>\n");
211 out
212}

268/// Die Sessions, Intent zuerst — neueste oben. (Absichtslose sind schon im
269/// [`Index`] ausgesiebt.)
270///
271/// Jede Karte ist ein Link auf die Datei-Seite, auf der die Session hängt; der
272/// Anker `#s-<id>` öffnet dort ihr Panel. Sessions ohne zugeordnete Datei-Seite
273/// (nichts Zugeordnetes zum Anklicken) bleiben eine schlichte Karte.
274fn session_list(index: &Index, session_page: &BTreeMap<SessionId, String>) -> String {
275 let mut items: Vec<(&SessionId, &Session)> = index.sessions().collect();
276 if items.is_empty() {
277 return "<p class=\"empty\">Keine Session mit erfasster Absicht.</p>\n".to_string();
278 }
279
280 // Neueste zuerst: nach Endzeitpunkt absteigend. Minds-Zeitstempel sind
281 // einheitlich RFC 3339 in UTC (…Z), deshalb ordnet der String-Vergleich
282 // chronologisch.
283 items.sort_by(|a, b| session_time(b.1).cmp(session_time(a.1)));
284
285 let mut out = String::from("<ul class=\"cards\">\n");
286 for (id, session) in items {
287 let summary = Summary::of(*id, session);
288 let guess = if index.is_observed(*id) {
289 ""
290 } else {
291 " · <span class=\"guess\">vermutet</span>"
292 };
293 let inner = format!(
294 "<p class=\"headline\">{headline}</p>\n\
295 <p class=\"sub\">{actor} · {files} Datei(en) · {input} ein / {output} aus Token{guess}</p>\n\
296 <p class=\"sub\"><code class=\"id\">{id}</code></p>\n",
297 headline = escape(&summary.headline),
298 actor = escape(&summary.actor),
299 files = summary.files,
300 input = summary.input_tokens,
301 output = summary.output_tokens,
302 id = escape(&summary.id.to_string()),
303 );
304
305 match session_page.get(id) {
306 Some(href) => out.push_str(&format!(
307 "<li class=\"card\"><a class=\"card-link\" href=\"{href}#s-{id}\">{inner}</a></li>\n",
308 href = escape(href),
309 id = escape(&id.to_string()),
310 )),
311 None => out.push_str(&format!("<li class=\"card\">{inner}</li>\n")),
312 }
313 }
314 out.push_str("</ul>\n");
315 out
316}

349/// Ein Dateiname für die Seite einer Datei.
350///
351/// Aus `src/retry.rs` wird `src-retry.rs.html`. Erlaubt bleiben nur
352/// `[A-Za-z0-9._-]`; alles andere — Schrägstriche, Leerzeichen, Umlaute — wird
353/// zu `-`. Damit ist der Name auf jedem Dateisystem und in jeder URL
354/// unverfänglich.
355///
356/// Die Abbildung ist **nicht** injektiv (`a/b` und `a-b` ergeben dasselbe); die
357/// Eindeutigkeit stellt der Aufrufer her, indem er Kollisionen durchnummeriert
358/// — siehe `render`.
359pub fn slug(path: &str) -> String {
360 let mut out = String::with_capacity(path.len() + 5);
361 for ch in path.chars() {
362 if ch.is_ascii_alphanumeric() || matches!(ch, '.' | '_' | '-') {
363 out.push(ch);
364 } else {
365 out.push('-');
366 }
367 }
368 if out.is_empty() {
369 out.push_str("datei");
370 }
371 out.push_str(".html");
372 out
373}

384/// Maskiert die fünf Zeichen, die in HTML-Text und -Attributen gefährlich sind.
385///
386/// Der Reader schreibt fremden Text in HTML: Prompts, Dateipfade, Modellnamen.
387/// Ohne Maskierung wäre ein Prompt mit `<script>` eine Lücke — und Prompts sind
388/// per Definition beliebiger Text. Deshalb geht **jeder** eingesetzte Wert hier
389/// durch, ausnahmslos.
390pub fn escape(text: &str) -> String {
391 let mut out = String::with_capacity(text.len());
392 for ch in text.chars() {
393 match ch {
394 '&' => out.push_str("&amp;"),
395 '<' => out.push_str("&lt;"),
396 '>' => out.push_str("&gt;"),
397 '"' => out.push_str("&quot;"),
398 '\'' => out.push_str("&#39;"),
399 other => out.push(other),
400 }
401 }
402 out
403}

458///
459/// Ohne JavaScript bleibt die Seite lesbar; dann sind alle Panels sichtbar,
460/// weil `hidden` erst hier gesetzt/entfernt wird.

463var panel=document.getElementById('panel');\
464if(!panel)return;\
465/* Erst hier verstecken: ohne Skript bleiben alle Panels lesbar. */\
466panel.querySelectorAll('.session').forEach(function(s){s.hidden=true});\
467function show(row){\
468var ids=(row.getAttribute('data-sessions')||'').split(' ').filter(Boolean);\
469panel.querySelectorAll('.session').forEach(function(s){s.hidden=true});\
470ids.forEach(function(id){var el=document.getElementById('s-'+id);if(el)el.hidden=false});\
471document.querySelectorAll('.line.selected').forEach(function(l){l.classList.remove('selected')});\

476var row=e.target.closest('.line.has-context');if(row)show(row);\
477});\

479if(e.key!=='Enter'&&e.key!==' ')return;\
480var row=e.target.closest&&e.target.closest('.line.has-context');\
