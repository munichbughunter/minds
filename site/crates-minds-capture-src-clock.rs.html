1//! Zeitstempel — RFC 3339 in UTC, ohne Datums-Dependency.
2//!
3//! Der Hook liegt auf dem heißen Pfad: Bei jedem Tool-Call startet unser
4//! Prozess neu. Eine Datums-Bibliothek dafür einzuziehen wäre nicht falsch,
5//! aber sie kostet Compile-Zeit und Binärgröße für eine einzige Zeile Ausgabe.
6//! Die Umrechnung „Sekunden seit Epoch → Kalenderdatum" ist ein geschlossenes,
7//! seit Jahrzehnten bekanntes Stück Arithmetik (Howard Hinnants
8//! `civil_from_days`); sie hier zu haben ist billiger als jede Alternative und
9//! vollständig testbar.
10//!
11//! # Wo Zeit *nicht* herkommt
12//!
13//! Nur der Hook liest die Uhr, weil nur er dabei ist, wenn etwas passiert. Der
14//! Adapter, der später aus dem Journal eine [`Session`](minds_core::Session)
15//! baut, ruft **niemals** [`now_rfc3339`] auf: Er würde damit bei jedem Lauf
16//! ein anderes Envelope und damit eine andere `SessionId` erzeugen. Derselbe
17//! Journal-Inhalt muss immer denselben Hash ergeben, sonst ist die
18//! Content-Adressierung eine Lüge und die Fixture-Tests aus M5 sind nicht
19//! schreibbar.
20
21use std::time::{SystemTime, UNIX_EPOCH};
22
23/// Jetzt, als RFC-3339-Zeitstempel in UTC mit Millisekunden
24/// (`2026-07-23T09:12:04.512Z`), zusammen mit den Nanosekunden seit Epoch.
25///
26/// Die zweite Zahl ist der Sortierschlüssel; die Zeichenkette ist für Menschen.
27/// Beide stammen aus **einer** Ablesung, damit sie nicht auseinanderlaufen.
28///
29/// Vor 1970 gibt es hier nichts zu tun: Liegt die Systemuhr davor, liefern wir
30/// `0`/Epoch statt zu panicken. Ein Hook darf nie die Sitzung des Nutzers
31/// abbrechen — auch nicht wegen einer kaputten Uhr.
32pub fn now() -> (String, u64) {
33 let nanos = SystemTime::now()
34 .duration_since(UNIX_EPOCH)
35 .map(|d| d.as_nanos())
36 .unwrap_or(0);
37 let nanos = u64::try_from(nanos).unwrap_or(u64::MAX);
38 (rfc3339_from_nanos(nanos), nanos)
39}
40
41/// Formatiert Nanosekunden seit Epoch als RFC 3339 in UTC mit Millisekunden.
42pub fn rfc3339_from_nanos(nanos: u64) -> String {
43 let secs = (nanos / 1_000_000_000) as i64;
44 let millis = (nanos % 1_000_000_000) / 1_000_000;
45
46 let days = secs.div_euclid(86_400);
47 let sod = secs.rem_euclid(86_400);
48 let (y, m, d) = civil_from_days(days);
49
50 format!(
51 "{y:04}-{m:02}-{d:02}T{:02}:{:02}:{:02}.{millis:03}Z",
52 sod / 3600,
53 (sod % 3600) / 60,
54 sod % 60,
55 )
56}
57
58/// Parst einen RFC-3339-Zeitstempel in UTC (`2026-07-23T09:12:04.512Z`) zu
59/// Sekunden seit Epoch.
60///
61/// Bewusst tolerant und ohne Datums-Dependency (dieselbe Linie wie der Rest
62/// dieses Moduls): Gelesen werden nur Datum und Uhrzeit auf Sekunden;
63/// Nachkommastellen, `Z` und ein Zeitzonen-Offset werden ignoriert. Die
64/// Transkripte, die das hier füttern, sind ausnahmslos UTC. Passt die Form
65/// nicht, ergibt es `None` statt eines geratenen Zeitpunkts.
66///
67/// Die Umkehrung von [`rfc3339_from_nanos`], für das Zuordnen importierter
68/// Sessions zu Commits ([`crate::match_commits`]).
69pub fn epoch_seconds_from_rfc3339(s: &str) -> Option<i64> {
70 let (date, time) = s.split_once('T')?;
71
72 let mut d = date.split('-');
73 let year: i64 = d.next()?.parse().ok()?;
74 let month: u32 = d.next()?.parse().ok()?;
75 let day: u32 = d.next()?.parse().ok()?;
76
77 // Nur `HH:MM:SS` — alles ab dem Punkt, `Z` oder `+`/`-` fällt weg.
78 let hms = &time[..time.find(['.', 'Z', '+']).unwrap_or(time.len())];
79 let mut t = hms.split(':');
80 let hour: i64 = t.next()?.parse().ok()?;
81 let min: i64 = t.next()?.parse().ok()?;
82 let sec: i64 = t.next().unwrap_or("0").parse().ok()?;
83
84 if !(1..=12).contains(&month) || !(1..=31).contains(&day) {
85 return None;
86 }
87
88 let days = days_from_civil(year, month, day);
89 Some(days * 86_400 + hour * 3_600 + min * 60 + sec)
90}

92/// (Jahr, Monat, Tag) → Tage seit 1970-01-01. Die Umkehrung von
93/// [`civil_from_days`], nach Howard Hinnant.
94fn days_from_civil(y: i64, m: u32, d: u32) -> i64 {
95 let y = y - i64::from(m <= 2);
96 let era = if y >= 0 { y } else { y - 399 } / 400;
97 let yoe = y - era * 400; // [0, 399]
98 let mp = if m > 2 { m - 3 } else { m + 9 } as i64; // [0, 11], 0 = März
99 let doy = (153 * mp + 2) / 5 + i64::from(d) - 1; // [0, 365]
100 let doe = yoe * 365 + yoe / 4 - yoe / 100 + doy; // [0, 146096]
101 era * 146_097 + doe - 719_468
102}
103
104/// Tage seit 1970-01-01 → (Jahr, Monat, Tag), proleptischer gregorianischer
105/// Kalender. Nach Howard Hinnant, `chrono`-Algorithmus ohne `chrono`.
106fn civil_from_days(z: i64) -> (i64, u32, u32) {
107 // Verschiebt den Nullpunkt auf 0000-03-01, damit der Schalttag ans
108 // Jahresende rutscht und keine Sonderfaelle mehr erzeugt.
109 let z = z + 719_468;
110 let era = if z >= 0 { z } else { z - 146_096 } / 146_097;
111 let doe = z - era * 146_097; // [0, 146096]
112 let yoe = (doe - doe / 1460 + doe / 36_524 - doe / 146_096) / 365; // [0, 399]
113 let y = yoe + era * 400;
114 let doy = doe - (365 * yoe + yoe / 4 - yoe / 100); // [0, 365]
115 let mp = (5 * doy + 2) / 153; // [0, 11], 0 = Maerz
116 let d = (doy - (153 * mp + 2) / 5 + 1) as u32; // [1, 31]
117 let m = if mp < 10 { mp + 3 } else { mp - 9 } as u32;
118
119 (y + i64::from(m <= 2), m, d)
120}

132 fn known_instants() {

Absicht

Ich möchte M5 so umbauen, dass wir es genauso machen wie entire! Mit dem Hook-Ansatz. Wir bleiben erst einmal bei der Rust implementierung. Diese ist effizienter. Das ist der entscheidende Unterschied. Minds soll auch Hooks installieren im Agenten selbst: Claude Code über `.claude/settings.json`, Codex über `.codex/hooks.json`, Cursor über `.cursor/hooks.json`, Gemini über `.gemini/settings.json`, OpenCode über ein TypeScript-Plugin. Bei Codex wird zusätzlich `codex_hooks = true` in die `config.toml` geschrieben. Dazu Git-Hooks: Checkpoints entstehen, wenn du oder der Agent committen; die Session-Metadaten möchte ich entweder auf einem child-repo haben oder falls wir diese kombination nicht haben direkt im repo. Das Verhalten soll an der Stelle absolut gleich sein.
