1//! `minds fsck` — hält der Record, was der Trailer verspricht?
2//!
3//! Der heiße Pfad ist fail-open: `minds hook` darf ein Event verlieren, statt
4//! die Sitzung zu stören. Der Preis dafür sind mögliche Lücken — und Lücken, die
5//! niemand sieht, sind schlimmer als keine. `fsck` macht sie sichtbar. Es prüft
6//! zwei Zusagen:
7//!
8//! 1. **Jeder Trailer ist auflösbar.** Zu jeder `Minds-Session-Id` in der
9//! Historie muss die Session im Store liegen. Ein Trailer, der ins Leere
10//! zeigt, ist eine Waise — der eine Integritätsbruch, den `fsck` mit einem
11//! Rückgabewert ≠ 0 quittiert.
12//! 2. **Das Journal ist heil.** Angesammelte, noch nicht eingecheckte Sessions
13//! werden gemeldet; Sequenzlücken (ein fail-open verlorenes Event) und
14//! beschädigte Dateien (ein abgestürzter Schreibvorgang) ebenso. Das sind
15//! Warnungen, kein Bruch: Sie erzählen, was der heiße Pfad gekostet hat.
16//!
17//! Ehrlich lückenhaft schlägt still vollständig — diese Datei ist die Einlösung
18//! dieses Satzes aus dem ganzen Entwurf.
19

31/// Führt `minds fsck` aus. Rückgabewert ≠ 0 genau dann, wenn ein Trailer nicht
32/// auflösbar ist.
33pub fn run() -> ExitCode {
34 match fsck() {
35 Ok(true) => ExitCode::SUCCESS,
36 Ok(false) => ExitCode::FAILURE,
37 Err(err) => {
38 eprintln!("minds fsck: {err}");
39 ExitCode::FAILURE
40 }
41 }
42}

44/// Gibt `true` zurück, wenn kein Trailer verwaist ist.
45fn fsck() -> Fallible<bool> {
46 let cwd = std::env::current_dir()?;
47 let repo = Repo::discover(&cwd)?;
48 let root = repo_root(&repo);
49 let store = config::load(&root).open(&root)?;
50
51 let orphans = check_trailers(&repo, store.as_ref())?;
52 let index_orphans = check_index(store.as_ref())?;
53 check_journal(&repo);
54
55 let total = orphans + index_orphans;
56 if total == 0 {
57 println!("fsck: in Ordnung");
58 Ok(true)
59 } else {
60 println!("fsck: {total} verwaiste(r) Verweis(e)");
61 Ok(false)
62 }
63}
64
65/// Prüft die Kanten des Store-Index: jede benannte Session muss im Store liegen.
66/// Gibt die Zahl der Waisen zurück.
67///
68/// Die Index-Kanten sind heuristisch ([`Evidence::Inferred`](minds_core::Evidence::Inferred)),
69/// aber die Session, auf die sie zeigen, muss trotzdem da sein — sonst ist der
70/// Verweis so tot wie ein verwaister Trailer.
71fn check_index(store: &dyn minds_store::ContextStore) -> Fallible<usize> {
72 let index = store.index()?;
73 if index.is_empty() {
74 println!("Index: leer");
75 return Ok(0);
76 }
77
78 let mut seen: BTreeSet<SessionId> = BTreeSet::new();
79 let mut orphans = 0usize;
80 let mut links = 0usize;
81
82 for (commit, entries) in index.iter() {
83 for entry in entries {
84 links += 1;
85 if !seen.insert(entry.session) {
86 continue;
87 }
88 if !store.exists(entry.session)? {
89 println!(" Waise: {commit} → {} (nicht im Store)", entry.session);
90 orphans += 1;
91 }
92 }
93 }
94
95 println!(
96 "Index: {links} vermutete Verknüpfung(en), {} eindeutig, {orphans} verwaist",
97 seen.len()
98 );
99 Ok(orphans)
100}

102/// Läuft die Historie ab HEAD ab und prüft jede eindeutige `Minds-Session-Id`
103/// gegen den Store. Gibt die Zahl der Waisen zurück.
104fn check_trailers(repo: &Repo, store: &dyn minds_store::ContextStore) -> Fallible<usize> {
105 let Some(head) = repo.head()?.commit() else {
106 println!("Trailer: HEAD hat noch keinen Commit — nichts zu prüfen");
107 return Ok(0);
108 };
109
110 // Eine Session-Id kann an mehreren Commits stehen (Rebase kopiert den
111 // Trailer). Jede nur einmal prüfen — der Store-Zugriff ist der teure Teil.
112 let mut seen: BTreeSet<SessionId> = BTreeSet::new();
113 let mut orphans = 0usize;
114 let mut total = 0usize;
115
116 for commit in repo.revwalk(head)? {
117 let commit = commit?;
118 for id in repo.session_ids_of(commit)? {
119 total += 1;
120 if !seen.insert(id) {
121 continue;
122 }
123 if !store.exists(id)? {
124 println!(" Waise: {commit} → {id} (nicht im Store)");
125 orphans += 1;
126 }
127 }
128 }
129
130 println!(
131 "Trailer: {total} Verweis(e), {} eindeutig, {orphans} verwaist",
132 seen.len()
133 );
134 Ok(orphans)
135}

137/// Meldet den Zustand des Journals: was noch aussteht, was fehlt, was beschädigt
138/// ist. Nur Warnungen — ein volles Journal ist der Normalfall zwischen zwei
139/// Commits.
140fn check_journal(repo: &Repo) {
141 let journal = Journal::open(repo.git_dir());
142 let Ok(sessions) = journal.sessions() else {
143 return;
144 };
145
146 if sessions.is_empty() {
147 println!("Journal: leer");
148 return;
149 }
150
151 println!(
152 "Journal: {} Session(s) noch nicht eingecheckt",
153 sessions.len()
154 );
155 for key in sessions {
156 let Ok(outcome) = journal.read(&key) else {
157 continue;
158 };
159 let mut notes = Vec::new();
160 if !outcome.gaps.is_empty() {
161 notes.push(format!("{} Lücke(n)", outcome.gaps.len()));
162 }
163 if !outcome.damaged.is_empty() {
164 notes.push(format!("{} beschädigt", outcome.damaged.len()));
165 }
166 let suffix = if notes.is_empty() {
167 String::new()
168 } else {
169 format!(" — {}", notes.join(", "))
170 };
171 println!(
172 " {}/{}: {} Event(s){suffix}",
173 key.agent(),
174 key.local_id(),
175 outcome.events.len()
176 );
177 }
178}

Das ist der entscheidende Unterschied. Minds soll auch Hooks installieren im Agenten selbst: Claude Code über `.claude/settings.json`, Codex über `.codex/hooks.json`, Cursor über `.cursor/hooks.json`, Gemini über `.gemini/settings.json`, OpenCode über ein TypeScript-Plugin. Bei Codex wird zusätzlich `codex_hooks = true` in die `config.toml` geschrieben. Dazu Git-Hooks: Checkpoints entstehen, wenn du oder der Agent committen; die Session-Metadaten möchte ich entweder auf einem child-repo haben oder falls wir diese kombination nicht haben direkt im repo. Das Verhalten soll an der Stelle absolut gleich sein.
