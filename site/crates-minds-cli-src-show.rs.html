1//! `minds show <commit>` — vom Commit zurück zur Absicht.
2//!
3//! Der erste Teil des Bug-Retrieval-Flows aus dem Plan, in einem Kommando:
4//!
5//! ```text
6//! Commit → Trailer lesen → Minds-Session-Id → Store auflösen → Intent zeigen
7//! ```
8//!
9//! Es liest nur. Kein Journal, kein Schreiben, keine Redaction — die ist längst
10//! passiert. Steht auf dem Commit kein Trailer, ist das kein Fehler, sondern die
11//! ehrliche Auskunft „für diesen Commit ist kein Kontext erfasst".
12//!
13//! # Was „Attribution" hier heißt
14//!
15//! Die deterministisch verfügbare Zuschreibung: welcher Agent, welches Modell,
16//! wie viele Token, welche Dateien, welche Kanten. Die zeilengenaue
17//! Mensch/Agent-Quote (`git blame` bis zum Prompt) ist eine eigene, größere
18//! Sache und gehört nicht in dieses schlanke Lese-Kommando.
19

28type Fallible<T> = std::result::Result<T, Box<dyn std::error::Error>>;

30/// Führt `minds show` aus. `rev` ist eine Git-Revision (`HEAD`, ein Hash, ein
31/// Tag); ohne Angabe wird HEAD gezeigt.
32pub fn run(rev: Option<&str>, full: bool) -> ExitCode {
33 match show(rev.unwrap_or("HEAD"), full) {
34 Ok(()) => ExitCode::SUCCESS,
35 Err(err) => {
36 eprintln!("minds show: {err}");
37 ExitCode::FAILURE
38 }
39 }
40}
41
42fn show(rev: &str, full: bool) -> Fallible<()> {
43 let cwd = std::env::current_dir()?;
44 let repo = Repo::discover(&cwd)?;
45 let root = repo_root(&repo);
46 let store = config::load(&root).open(&root)?;
47
48 let commit = resolve(&root, rev).ok_or_else(|| format!("keine solche Revision: {rev}"))?;
49
50 // Trailer (beobachtet) und Store-Index (vermutet) zusammenführen — so
51 // erscheinen auch importierte Sessions, die keinen Trailer haben.
52 let trailers = repo.session_ids_of(commit)?;
53 let index = store.index()?;
54 let links = render::merge_links(&trailers, index.links_of(&commit.to_string()));
55
56 crate::render::show_links(&format!("commit {commit}"), &links, store.as_ref(), full)
57}
58
59/// Löst eine Git-Revision zu einem vollen Commit-Hash auf — über `git
60/// rev-parse`, weil das jede Schreibweise versteht (HEAD, Tags, Kurzhashes,
61/// `HEAD~2`), die selbst zu implementieren müßig wäre.
62fn resolve(root: &Path, rev: &str) -> Option<CommitId> {
63 let output = Command::new("git")
64 .arg("-C")
65 .arg(root)
66 .args([
67 "rev-parse",
68 "--verify",
69 "--quiet",
70 &format!("{rev}^{{commit}}"),
71 ])
72 .output()
73 .ok()?;
74 if !output.status.success() {
75 return None;
76 }
77 String::from_utf8_lossy(&output.stdout).trim().parse().ok()
78}

80fn repo_root(repo: &Repo) -> std::path::PathBuf {
81 repo.git_dir()
82 .parent()
83 .unwrap_or_else(|| repo.git_dir())
84 .to_path_buf()
85}
