1//! Das `minds`-Binary.
2//!
3//! Die Kommandos: [`enable`] (Setup — Hooks + Store-Config), [`hook`] (heißer
4//! Pfad), `checkpoint`/`show`/`why`/`fsck` (kalter Pfad, M6).
5//!
6//! # Warum hier kein clap steht
7//!
8//! Ein Argument-Parser ist eine Aussage über die Kommandostruktur. Die
9//! Kommandostruktur von Minds ist noch nicht entworfen; sie jetzt an einem
10//! einzigen Unterkommando auszurichten hieße, sie in M6 wieder umzubauen. Der
11//! Parser unten ist deshalb absichtlich zu dumm, um zu bleiben: kein
12//! Subkommando-Baum, keine Kurzformen, keine Gruppierung.
13//!
14//! Der zweite Grund ist der heiße Pfad. `minds hook` startet bei jedem
15//! Tool-Call des Agenten neu — der Prozess soll stdin lesen, eine Datei
16//! schreiben und enden.
17//!
18//! # Die Ausnahme bei den Rückgabewerten
19//!
20//! Alle künftigen Unterkommandos melden Fehler über den Rückgabewert, wie es
21//! sich gehört. `hook` nicht: Es endet **immer** mit 0, auch bei falschen
22//! Argumenten. Der Grund steht in [`hook`] — bei Claude Code bedeutet Exit 2
23//! „blockiere diese Aktion", und ein Rekorder, der wegen eines fehlenden
24//! `--agent` die Arbeit des Nutzers stoppt, hat seinen Zweck verfehlt.
25//!
26//! Der Preis ist, dass eine kaputte Hook-Konfiguration still bleibt. Deshalb
27//! landet sie in `<git-dir>/minds/hook.log`, und deshalb wird `minds fsck` in
28//! M6 danach sehen.

85fn main() -> ExitCode {
86 let args: Vec<String> = std::env::args().skip(1).collect();
87
88 match args.first().map(String::as_str) {
89 Some("enable") => {
90 // Verstecktes internes Flag: `minds enable` startet den Backfill als
91 // losgelösten Hintergrundprozess, der sich selbst hiermit aufruft.
92 // Kein öffentliches `minds import` — der Nutzer sieht davon nichts
93 // (steht bewusst nicht in USAGE).
94 if args.iter().any(|a| a == enable::BACKGROUND_IMPORT_FLAG) {
95 import_cmd::run()
96 } else {
97 let agent = flag(&args, "--agent");
98 let store = store_config_from(&args);
99 let child_remote = flag(&args, "--child-remote");
100 let verbose = args.iter().any(|a| a == "-v" || a == "--verbose");
101 enable::run(agent.as_deref(), &store, child_remote.as_deref(), verbose)
102 }
103 }
104
105 Some("checkpoint") => {
106 let commit = flag(&args, "--commit");
107 checkpoint::run(commit.as_deref())
108 }
109
110 Some("show") => {
111 let full = args.iter().any(|a| a == "--full");
112 show::run(positional(&args), full)
113 }
114
115 Some("why") => {
116 let full = args.iter().any(|a| a == "--full");
117 why::run(positional(&args), full)
118 }
119
120 Some("fsck") => fsck::run(),
121
122 Some("render") => {
123 let out = flag(&args, "--out");
124 render_cmd::run(out.as_deref())
125 }
126
127 Some("hook") => {
128 let agent = flag(&args, "--agent");
129 let event = flag(&args, "--event");
130 hook::run(agent.as_deref(), event.as_deref())
131 }
132
133 Some("--version" | "-V") => {
134 println!("minds {}", env!("CARGO_PKG_VERSION"));
135 ExitCode::SUCCESS
136 }
137
138 Some("--help" | "-h") | None => {
139 print!("{USAGE}");
140 ExitCode::SUCCESS
141 }
142
143 Some(other) => {
144 eprintln!("unbekanntes Unterkommando: {other}\n");
145 eprint!("{USAGE}");
146 ExitCode::FAILURE
147 }
148 }
149}

151/// Das erste positionale Argument nach dem Unterkommando (das erste, das nicht
152/// mit `-` beginnt). So darf `--full` vor *oder* nach dem Commit stehen.
153fn positional(args: &[String]) -> Option<&str> {
154 args.iter()
155 .skip(1)
156 .find(|a| !a.starts_with('-'))
157 .map(String::as_str)
158}
159
160/// Liest `--name wert` aus den Argumenten.
161///
162/// Bewusst ohne `--name=wert`, ohne Kurzformen und ohne Fehlermeldung bei
163/// Unbekanntem: ein absichtlich dummer Parser, solange die Kommandostruktur
164/// jung ist.
165fn flag(args: &[String], name: &str) -> Option<String> {
166 args.iter()
167 .position(|a| a == name)
168 .and_then(|i| args.get(i + 1))
169 .cloned()
170}
171
172/// Baut die [`StoreConfig`] aus `--child-repo`/`--ref`. Ohne Flags: In-Repo mit
173/// Default-Ref — die Einstellung, für die niemand etwas tun muss.
174fn store_config_from(args: &[String]) -> StoreConfig {
175 let base = match flag(args, "--child-repo") {
176 Some(path) => StoreConfig::child_repo(path),
177 None => StoreConfig::in_repo(),
178 };
179 match flag(args, "--ref") {
180 Some(reference) => base.with_ref(reference),
181 None => base,
182 }
183}

Ich möchte M5 so umbauen, dass wir es genauso machen wie entire! Mit dem Hook-Ansatz. Wir bleiben erst einmal bei der Rust implementierung. Diese ist effizienter. Das ist der entscheidende Unterschied. Dazu Git-Hooks: Checkpoints entstehen, wenn du oder der Agent committen; die Session-Metadaten möchte ich entweder auf einem child-repo haben oder falls wir diese kombination nicht haben direkt im repo. Das Verhalten soll an der Stelle absolut gleich sein.
