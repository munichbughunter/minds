19/// Führt `minds render` aus.
20pub fn run(out: Option<&str>) -> ExitCode {
21 match render(Path::new(out.unwrap_or(DEFAULT_OUT))) {
22 Ok(()) => ExitCode::SUCCESS,
23 Err(err) => {
24 eprintln!("minds render: {err}");
25 ExitCode::FAILURE

30fn render(out: &Path) -> Fallible<()> {
31 let cwd = std::env::current_dir()?;
32 let repo = Repo::discover(&cwd)?;
33 let root = repo_root(&repo);
34 let store = config::load(&root).open(&root)?;
35
36 let site = minds_reader::render(&repo, store.as_ref(), out)?;
37
38 println!(
39 " {} Datei(en), {} Session(s) → {}",
40 site.files,
41 site.sessions,
42 site.out.display()
43 );
44 if site.skipped > 0 {
45 println!(
46 " {} Datei(en) übersprungen (kein UTF-8 oder Blame nicht möglich)",
47 site.skipped
48 );
49 }
50 println!(" öffnen: {}", site.out.join("index.html").display());
51 Ok(())
52}

Ich möchte M5 so umbauen, dass wir es genauso machen wie entire! Mit dem Hook-Ansatz. Wir bleiben erst einmal bei der Rust implementierung. Diese ist effizienter. Das Verhalten soll an der Stelle absolut gleich sein.
