1//! Eine Datei, Zeile für Zeile mit ihrer Session verbunden — der Kern des
2//! Magic Moments.
3//!
4//! ```text
5//! Zeile ──blame──► Commit ──Trailer──► Session
6//! ```
7//!
8//! [`FileView::join`] ist eine **reine Funktion**: Dateiinhalt rein, Blame rein,
9//! [`Index`] rein, fertige Ansicht raus. Kein Git, kein Dateisystem. Das ist
10//! Absicht — die interessante Logik (welche Zeile gehört zu welcher Session)
11//! lässt sich damit ohne Repository prüfen, und der Aufrufer entscheidet, woher
12//! Inhalt und Blame kommen.
13//!
14//! # Nicht jede Zeile hat eine Session
15//!
16//! Zeilen aus Commits ohne Trailer — alles, was vor Minds entstand oder von Hand
17//! geschrieben wurde — tragen schlicht keine. Das ist der Normalfall und wird
18//! nicht als Lücke dargestellt: Der Reader zeigt, was belegt ist, und behauptet
19//! nichts über den Rest.
20

28/// Eine Datei mit ihren Zeilen und deren Zuordnung.
29#[derive(Debug, Clone, PartialEq, Eq)]
30pub struct FileView {
31 /// Repo-relativer Pfad.
32 pub path: String,
33 /// Die Zeilen, 1-basiert nummeriert und in Dateireihenfolge.
34 pub lines: Vec<Line>,
35}

37/// Eine einzelne Zeile samt Herkunft.
38#[derive(Debug, Clone, PartialEq, Eq)]
39pub struct Line {
40 /// Zeilennummer, **1-basiert** — wie `git blame` und wie ein Editor zählt.
41 pub number: u32,
42 /// Der Text der Zeile, ohne Zeilenumbruch.
43 pub text: String,
44 /// Der Commit, der sie zuletzt geändert hat; `None`, wenn der Blame sie
45 /// nicht kennt.
46 pub commit: Option<CommitId>,
47 /// Die Sessions hinter diesem Commit. Leer heißt „kein erfasster Kontext",
48 /// nicht „unbekannt".
49 pub sessions: Vec<SessionId>,
50}
51
52impl Line {
53 /// `true`, wenn hinter dieser Zeile mindestens eine Session steht.
54 pub fn is_attributed(&self) -> bool {
55 !self.sessions.is_empty()
56 }
57}

59impl FileView {
60 /// Verbindet Dateiinhalt, Blame und Index zu einer Ansicht.
61 ///
62 /// `content` wird an Zeilenumbrüchen zerlegt; `blame` darf lückenhaft oder
63 /// unsortiert sein — gesucht wird über die Zeilennummer, nicht über die
64 /// Position.
65 pub fn join(path: &str, content: &str, blame: &[BlameLine], index: &Index) -> Self {
66 let by_line: BTreeMap<u32, CommitId> = blame
67 .iter()
68 .map(|entry| (entry.line, entry.commit))
69 .collect();
70
71 let lines = content
72 .lines()
73 .enumerate()
74 .map(|(offset, text)| {
75 // `enumerate` zählt ab 0, Blame und Editor ab 1.
76 let number = offset as u32 + 1;
77 let commit = by_line.get(&number).copied();
78 let sessions = commit
79 .map(|commit| index.sessions_of(commit).to_vec())
80 .unwrap_or_default();
81 Line {
82 number,
83 text: text.to_string(),
84 commit,
85 sessions,
86 }
87 })
88 .collect();
89
90 Self {
91 path: path.to_string(),
92 lines,
93 }
94 }
95
96 /// Wie viele Zeilen mindestens eine Session tragen.
97 pub fn attributed_lines(&self) -> usize {
98 self.lines
99 .iter()
100 .filter(|line| line.is_attributed())
101 .count()
102 }
103
104 /// `true`, wenn irgendeine Zeile eine Session trägt — nur solche Dateien
105 /// sind für den Reader interessant.
106 pub fn is_attributed(&self) -> bool {
107 self.lines.iter().any(Line::is_attributed)
108 }
109
110 /// Die Sessions, die diese Datei berühren, in der Reihenfolge ihres ersten
111 /// Auftretens und ohne Wiederholung.
112 pub fn sessions(&self) -> Vec<SessionId> {
113 let mut seen = Vec::new();
114 for id in self.lines.iter().flat_map(|line| &line.sessions) {
115 if !seen.contains(id) {
116 seen.push(*id);
117 }
118 }
119 seen
120 }
121}

157 /// Commit '1' trägt Session 'a', Commit '2' trägt 'a' und 'b'.
158 fn index() -> Index {
159 let mut sessions = BTreeMap::new();
160 sessions.insert(sid('a'), session());
161 sessions.insert(sid('b'), session());
162 let mut commits = BTreeMap::new();
163 commits.insert(cid('1'), vec![sid('a')]);
164 commits.insert(cid('2'), vec![sid('a'), sid('b')]);
165 Index::from_parts(sessions, commits)
