1//! Backfill: bestehende Agent-Transkripte in [`Session`]s, für die Zeit **vor**
2//! der Einrichtung.
3//!
4//! ADR-0003 hat die Live-Erfassung auf Hooks umgestellt; ADR-0004 ergänzt sie um
5//! diesen einmaligen Import. Der Hook bleibt der Normalweg — dieses Modul erntet
6//! nur nach, was schon geschah, bevor `minds enable` lief, und was die Agents
7//! noch als Transkript auf der Platte halten.
8//!
9//! # Ehrlich zu den Formaten
10//!
11//! Claude Code hat einen echten Reader (`~/.claude/projects/<slug>/<id>.jsonl`,
12//! Format bekannt). Für Codex, Cursor, Gemini und OpenCode kennen wir Ort und
13//! Form nicht belastbar; ihre Reader sind Platzhalter, die **nichts** liefern
14//! und das sagen, statt Geratenes abzulegen. Das ist dieselbe Linie wie überall:
15//! lieber eine ehrliche Lücke als ein stiller Fehler.
16//!
17//! # Was ein importierter Record *nicht* ist
18//!
19//! Eine importierte Session ist deterministisch aus dem Transkript gebaut — aber
20//! sie hat nie ein Journal gesehen. Die Ordnung über Agents hinweg, die der
21//! Hook-Weg beobachtet, fehlt ihr; ihre Verknüpfung zu einem Commit ist später
22//! eine *Vermutung* (siehe `crate::match_commits`), kein beobachteter Trailer.
23//! Der Import ist eine gute Näherung, keine Aufzeichnung.

74/// Sucht die Claude-Code-Transkripte für `repo_root` und baut je Datei eine
75/// Session.
76///
77/// Claude Code legt Transkripte unter `~/.claude/projects/<slug>/<id>.jsonl` ab,
78/// wobei `<slug>` das Arbeitsverzeichnis mit `/` und `.` zu `-` ist. Existiert
79/// das Verzeichnis nicht, hat der Nutzer mit Claude Code in diesem Repo nie
80/// gearbeitet — kein Fehler, nur nichts zu tun.
81fn import_claude_code(repo_root: &Path, home: &Path) -> AgentImport {
82 let dir = home
83 .join(".claude")
84 .join("projects")
85 .join(claude_slug(repo_root));
86
87 let mut sessions = Vec::new();
88 let mut note = None;
89
90 match std::fs::read_dir(&dir) {
91 Ok(entries) => {
92 let mut files: Vec<PathBuf> = entries
93 .flatten()
94 .map(|e| e.path())
95 .filter(|p| p.extension().is_some_and(|e| e == "jsonl"))
96 .collect();
97 // Sortiert, damit die Reihenfolge nicht von der Platte abhängt.
98 files.sort();
99 for path in files {
100 let fallback = path
101 .file_stem()
102 .and_then(|s| s.to_str())
103 .unwrap_or("unbekannt");
104 match std::fs::read(&path) {
105 Ok(bytes) => sessions.extend(parse_claude_code(&bytes, fallback)),
106 Err(err) => note = Some(format!("{} nicht lesbar: {err}", path.display())),
107 }
108 }
109 }
110 Err(err) if err.kind() == std::io::ErrorKind::NotFound => {}
111 Err(err) => note = Some(format!("{} nicht lesbar: {err}", dir.display())),
112 }
113
114 AgentImport {
115 agent: "claude-code".to_string(),
116 sessions,
117 note,
118 }
119}

121/// Claude Codes Verzeichnisname für ein Arbeitsverzeichnis: jeder `/` und `.`
122/// wird zu `-`.
123fn claude_slug(repo_root: &Path) -> String {
124 repo_root
125 .to_string_lossy()
126 .chars()
127 .map(|c| if c == '/' || c == '.' { '-' } else { c })
128 .collect()
129}

131/// Baut aus einem Claude-Code-Transkript (JSONL) **eine Session je `git
132/// commit`** — segmentiert, statt den ganzen Verlauf zu einem Klumpen zu machen.
133///
134/// # Warum segmentieren
135///
136/// Ein Claude-Code-Transkript ist *eine* Sitzung, die oft viele Commits
137/// hervorbringt. Als eine einzige Session gebaut, hinge sie an jedem dieser
138/// Commits — `minds show <commit>` zeigte für jeden dasselbe. Deshalb wird der
139/// Verlauf an den Stellen geschnitten, an denen der Agent `git commit` ausführt:
140/// Jedes Segment ist die Arbeit *bis zu und einschließlich* eines Commits, wird
141/// eine eigene Session und matcht über seine Dateien und sein (kleines)
142/// Zeitfenster genau den Commit, den es erzeugt hat.
143///
144/// Deterministisch: keine Uhr, kein Zufall. Dieselben Bytes ergeben dieselben
145/// Sessions und damit dieselben `SessionId`s.
146pub fn parse_claude_code(bytes: &[u8], fallback_local_id: &str) -> Vec<Session> {
147 let text = String::from_utf8_lossy(bytes);
148
149 // Über den ganzen Verlauf konstant: Modell, Version, Kennung, cwd.
150 let mut model: Option<String> = None;
151 let mut version: Option<String> = None;
152 let mut session_id: Option<String> = None;
153 let mut cwd: Option<String> = None;
154
155 let mut segments: Vec<Segment> = Vec::new();
156 let mut cur = Segment::default();
157 // Der zuletzt gesehene Prompt. Ein Prompt kann zu mehreren Commits führen;
158 // die Folge-Segmente haben dann keinen *eigenen* Prompt und erben diesen —
159 // sonst stünde dort „(kein Prompt erfasst)", obwohl die Absicht bekannt ist.
160 let mut carried: Option<String> = None;
161
162 for line in text.lines() {
163 let line = line.trim();
164 if line.is_empty() {
165 continue;
166 }
167 let Ok(entry) = serde_json::from_str::<Line>(line) else {
168 continue;
169 };
170
171 session_id = session_id.or(entry.session_id);
172 version = version.or(entry.version);
173 cwd = cwd.or(entry.cwd.clone());
174
175 let Some(message) = entry.message else {
176 continue;
177 };
178
179 match entry.kind.as_deref() {
180 Some("user") => {
181 // Ein User-*String* ist ein Prompt; eine Liste sind Tool-Ergebnisse,
182 // deren Inhalt wir nicht aufheben (der Effekt steht am Tool-Call).
183 if let Some(serde_json::Value::String(prompt)) = message.content {
184 if !prompt.trim().is_empty() {
185 cur.note_time(&entry.timestamp);
186 cur.request.get_or_insert_with(|| prompt.clone());
187 carried = Some(prompt.clone());
188 cur.turns.push(user_turn(prompt, entry.timestamp.clone()));
189 }
190 }
191 }
192 Some("assistant") => {
193 model = model.or(message.model);
194 if let Some(usage) = message.usage {
195 cur.input = cur.input.saturating_add(usage.input_tokens);
196 cur.output = cur.output.saturating_add(usage.output_tokens);
197 }
198 cur.note_time(&entry.timestamp);
199 let (text, tool_calls) = assistant_blocks(message.content.as_ref());
200 let commits = tool_calls.iter().any(is_git_commit);
201 cur.turns.push(Turn {
202 role: minds_core::Role::Assistant,
203 text,
204 tool_calls,
205 parent: None,
206 at: entry.timestamp.clone(),
207 });
208 // Der Commit schließt das Segment ab — die nächste Arbeit gehört
209 // zu einem anderen Commit.
210 if commits {
211 finalize(&mut segments, std::mem::take(&mut cur), &carried);
212 }
213 }
214 _ => {}
215 }
216 }
217 finalize(&mut segments, cur, &carried);
218
219 let model = model
220 .map(|id| Model {
221 provider: transcript::provider_of(&id),
222 id,
223 })
224 .unwrap_or_else(|| Model {
225 provider: unknown(),
226 id: unknown(),
227 });
228 let version = version.unwrap_or_else(unknown);
229 let local_id = session_id.unwrap_or_else(|| fallback_local_id.to_string());
230
231 segments
232 .into_iter()
233 .map(|seg| seg.into_session(&version, &model, &local_id, cwd.as_deref()))
234 .collect()
235}
236
237/// Schließt ein Segment ab: Fehlt ihm ein eigener Prompt, erbt es den zuletzt
238/// gesehenen (`carried`). Ohne jede Absicht — nur die Segmente vor dem ersten
239/// Prompt — wird es verworfen, denn ein Record ohne Absicht ist keiner.
240fn finalize(segments: &mut Vec<Segment>, mut seg: Segment, carried: &Option<String>) {
241 if seg.turns.is_empty() {
242 return;
243 }
244 if seg.request.is_none() {
245 seg.request = carried.clone();
246 }
247 if seg.request.is_some() {
248 segments.push(seg);
249 }
250}

265 fn note_time(&mut self, ts: &Option<String>) {
266 if let Some(ts) = ts {

274 /// Baut die Session dieses Segments; die konstanten Metadaten kommen von
275 /// außen.
276 fn into_session(
277 self,
278 version: &str,
279 model: &Model,
280 local_id: &str,
281 cwd: Option<&str>,
282 ) -> Session {
283 let files = written_files(&self.turns, cwd);
284 Session {
285 schema_version: minds_core::SCHEMA_VERSION,
286 agent: Agent {
287 name: "claude-code".to_string(),
288 version: version.to_string(),
289 },
290 model: model.clone(),
291 intent: Intent {
292 request: self.request.unwrap_or_default(),
293 ..Intent::default()
294 },
295 turns: self.turns,
296 usage: minds_core::Usage {
297 input_tokens: self.input,
298 output_tokens: self.output,
299 },
300 produced: Produced {
301 commit_hint: None,
302 files,
303 },
304 redaction: minds_core::Redaction::default(),
305 lineage: Some(Lineage {
306 local_id: local_id.to_string(),
307 started_at: self.first_ts,
308 ended_at: self.last_ts,
309 cwd: cwd.map(str::to_string),
310 }),
311 edges: Vec::new(),
312 }
313 }

316/// Ob ein Tool-Call ein `git commit` ist — die Segmentgrenze. Geprüft am rohen

337/// Zerlegt den Assistant-Inhalt in Antworttext und Tool-Calls.
338fn assistant_blocks(content: Option<&serde_json::Value>) -> (String, Vec<ToolCall>) {
339 let Some(serde_json::Value::Array(blocks)) = content else {
340 // Ein schlichter String ist selten, aber möglich.
341 if let Some(serde_json::Value::String(s)) = content {
342 return (s.clone(), Vec::new());
343 }
344 return (String::new(), Vec::new());
345 };
346
347 let mut text_parts = Vec::new();
348 let mut tool_calls = Vec::new();
349
350 for block in blocks {
351 match block.get("type").and_then(|t| t.as_str()) {
352 Some("text") => {
353 if let Some(t) = block.get("text").and_then(|t| t.as_str()) {
354 text_parts.push(t.to_string());
355 }
356 }
357 Some("tool_use") => {
358 let name = block
359 .get("name")
360 .and_then(|n| n.as_str())
361 .unwrap_or("")
362 .to_string();
363 let input = block.get("input");
364 let arguments = input.map(|i| i.to_string()).unwrap_or_default();
365 let raw = input.and_then(|i| RawValue::from_string(i.to_string()).ok());
366 let effect = normalize::claude_effect(&name, raw.as_deref());
367 tool_calls.push(ToolCall {
368 name,
369 arguments,
370 effect: Some(effect),
371 });
372 }
373 _ => {}
374 }
375 }
376
377 (text_parts.join("\n"), tool_calls)
378}

380/// Die von der Session geschriebenen Dateien, repo-relativ (gegen `cwd`
381/// gekürzt), sortiert und ohne Duplikate.
382fn written_files(turns: &[Turn], cwd: Option<&str>) -> Vec<String> {
383 let mut files: Vec<String> = turns
384 .iter()
385 .flat_map(|t| &t.tool_calls)
386 .filter_map(|c| c.effect.as_ref())
387 .filter(|e| matches!(e.kind, EffectKind::Write | EffectKind::Delete))
388 .filter_map(|e| e.path.as_deref())
389 .map(|path| relativize(path, cwd))
390 .collect();
391 files.sort();
392 files.dedup();
393 files
394}

396/// Kürzt einen absoluten Pfad gegen `cwd`, damit er repo-relativ wird (wie Git
397/// ihn führt). Passt `cwd` nicht, bleibt der Pfad, wie er ist.
398fn relativize(path: &str, cwd: Option<&str>) -> String {
399 if let Some(cwd) = cwd {
400 let prefix = format!("{}/", cwd.trim_end_matches('/'));
401 if let Some(rest) = path.strip_prefix(&prefix) {
402 return rest.to_string();
403 }
404 }
405 path.to_string()
406}
