1//! Die Store-Config, wo sie hingehört: in `.git/config`.
2//!
3//! [`minds_store::StoreConfig`] ist ein Wert ohne Dateiformat — bewusst, denn wo
4//! er liegt, entscheidet die CLI. Der naheliegende Ort ist `.git/config`
5//! (`minds.backend`, `minds.contextRef`, `minds.childPath`): keine neue Datei,
6//! pro Klon gültig, von Git verwaltet, überlebt jeden Pull. `minds enable`
7//! schreibt ihn, die übrigen Kommandos lesen ihn.
8//!
9//! # Warum über `git config` und nicht über einen INI-Parser
10//!
11//! `.git/config` ist INI mit Feinheiten (Sub-Sections, `include`, Groß-/Klein-
12//! schreibung). Es selbst zu parsen hieße, einen Teil von Git nachzubauen und
13//! dabei falsch zu liegen. Stattdessen rufen wir `git config` auf — dieselbe
14//! pragmatische Linie wie [`minds_git::ShellBlame`]: Git ist ohnehin die eine
15//! harte Abhängigkeit, also darf die CLI es benutzen, wo die Bibliothek (noch)
16//! keine Schnittstelle bietet.
17//!
18//! # Fehlt die Config, ist das kein Fehler
19//!
20//! [`load`] fällt auf den Default zurück (In-Repo, `refs/minds/context`). Ein
21//! Repo, in dem `minds enable` nie lief, ist damit trotzdem lesbar — nur eben
22//! mit den Standardwerten. Fail-open beim Lesen, streng erst beim Schreiben.

30/// Schlüssel in `.git/config`.
31const KEY_BACKEND: &str = "minds.backend";
32const KEY_REF: &str = "minds.contextRef";
33const KEY_CHILD_PATH: &str = "minds.childPath";
34
35/// Wert von [`KEY_BACKEND`] für die beiden Backends.
36const BACKEND_IN_REPO: &str = "in-repo";
37const BACKEND_CHILD: &str = "child-repo";
38
39/// Schreibt die Config nach `.git/config` unter `repo_root`.
40///
41/// Idempotent von Natur aus: `git config --local <key> <value>` setzt oder
42/// ersetzt, es dupliziert nicht.
43pub fn write(repo_root: &Path, config: &StoreConfig) -> std::io::Result<()> {
44 match config.backend() {
45 Backend::InRepo => {
46 set(repo_root, KEY_BACKEND, BACKEND_IN_REPO)?;
47 }
48 Backend::ChildRepo { path } => {
49 set(repo_root, KEY_BACKEND, BACKEND_CHILD)?;
50 set(repo_root, KEY_CHILD_PATH, &path.to_string_lossy())?;
51 }
52 }
53 set(repo_root, KEY_REF, config.reference())?;
54 Ok(())
55}
56
57/// Liest die Config aus `.git/config`. Fehlt sie ganz oder teilweise, greifen
58/// die Defaults (In-Repo, [`DEFAULT_CONTEXT_REF`]).
59pub fn load(repo_root: &Path) -> StoreConfig {
60 let reference = get(repo_root, KEY_REF).unwrap_or_else(|| DEFAULT_CONTEXT_REF.to_string());
61
62 match get(repo_root, KEY_BACKEND).as_deref() {
63 Some(BACKEND_CHILD) => {
64 let path = get(repo_root, KEY_CHILD_PATH).unwrap_or_default();
65 StoreConfig::child_repo(path).with_ref(reference)
66 }
67 _ => StoreConfig::in_repo().with_ref(reference),
68 }
69}

71fn set(repo_root: &Path, key: &str, value: &str) -> std::io::Result<()> {
72 let status = Command::new("git")
73 .arg("-C")
74 .arg(repo_root)
75 .args(["config", "--local", key, value])
76 .status()?;
77 if status.success() {
78 Ok(())
79 } else {
80 Err(std::io::Error::other(format!(
81 "`git config --local {key}` schlug fehl"
82 )))
83 }
84}
85
86fn get(repo_root: &Path, key: &str) -> Option<String> {
87 let output = Command::new("git")
88 .arg("-C")
89 .arg(repo_root)
90 .args(["config", "--local", "--get", key])
91 .output()
92 .ok()?;
93 if !output.status.success() {
94 return None;
95 }
96 let value = String::from_utf8_lossy(&output.stdout).trim().to_string();
97 if value.is_empty() { None } else { Some(value) }
98}

Absicht

Ich möchte M5 so umbauen, dass wir es genauso machen wie entire! Mit dem Hook-Ansatz. Wir bleiben erst einmal bei der Rust implementierung. Diese ist effizienter. Das ist der entscheidende Unterschied. Minds soll auch Hooks installieren im Agenten selbst: Claude Code über `.claude/settings.json`, Codex über `.codex/hooks.json`, Cursor über `.cursor/hooks.json`, Gemini über `.gemini/settings.json`, OpenCode über ein TypeScript-Plugin. Bei Codex wird zusätzlich `codex_hooks = true` in die `config.toml` geschrieben. Dazu Git-Hooks: Checkpoints entstehen, wenn du oder der Agent committen; die Session-Metadaten möchte ich entweder auf einem child-repo haben oder falls wir diese kombination nicht haben direkt im repo. Das Verhalten soll an der Stelle absolut gleich sein.
